
    // C3 ratificada: los MSS de C2 se heredan, los faltantes se esperan en C3.
    var bool c3Active = false
    var int c3Start = na
    var int c1Start3 = na
    var float c1High3 = na
    var float c1Low3 = na
    var int c3Dir = 0
    var int c3SweepEnd = na
    var float c3Pivot5 = na
    var float c3Pivot15 = na
    var int c3Pivot5At = na
    var int c3Pivot15At = na
    var int c3M5At = na
    var int c3M15At = na
    var float c3Invalidation = na
    // Metadatos del CRT ratificado que se encuentra en C3.
    var string c3ModelKind = "NONE"
    var bool c3PatternValid = false
    var int c3ModelC1 = na
    var int c3C2Start = na // Solo metadato para asociar PDH/PDL, sin alterar MSS.

    // Clasificacion diagnostica del ultimo cierre C2, aun si no es operativo.
    var string last2CClassification = "NONE"
    var int last2CClassificationAt = na

    // Memoria controlada del ULTIMO CRT realmente ratificado, NO de
    // cualquier MSS aislado. Los flags se copian desde C2/C3 y no se
    // inventan desde eventos antiguos no vinculados al ciclo.
    var int lastC1 = na
    var int lastDir = 0
    var int lastConfirmedAt = na
    var float lastHi = na
    var float lastLo = na
    var int lastSweepAt = na
    var int lastM5 = na
    var int lastM15 = na
    var float lastM5Level = na
    var float lastM15Level = na
    var int lastM5PivotAt = na
    var int lastM15PivotAt = na
    var string lastModelKind = "NONE"
    var bool lastPatternValid = false
    var int lastModelC1 = na
    var int lastC2Start = na // Memoria del mismo CRT ratificado.

    // Procesamos la ULTIMA M5 cerrada, no la M5 en formacion.
    // f_engine() debe evaluarse incondicionalmente en cada barra M5.
    int bucket15 = time("15")[1]
    int bucketH4 = isIndex ? time("240", "0200-0200:1234567", TZ)[1] : time("240", "0100-0100:1234567", TZ)[1]

    // ---- CERRAR M15 ANTERIOR ----
    if not na(barT) and not na(bucket15)
        bool next15 = not na(m15Bucket) and bucket15 != m15Bucket
        if next15
            if m15Count == 3 and m15LastEnd == bucket15
                array.push(highs15, m15High)
                array.push(lows15, m15Low)
                array.push(times15, m15Start)
                if array.size(highs15) > 48
                    array.shift(highs15)
                    array.shift(lows15)
                    array.shift(times15)
                int n = array.size(highs15)
                int pivot = n - 1 - right15
                if pivot >= left15 and pivot >= 0
                    float candH = array.get(highs15, pivot)
                    float candL = array.get(lows15, pivot)
                    bool okHi = true
                    bool okLo = true
                    for j = -left15 to right15
                        if j != 0
                            okHi := okHi and candH > array.get(highs15, pivot + j)
                            okLo := okLo and candL < array.get(lows15, pivot + j)
                    if okHi
                        lastHigh15 := candH
                        lastHigh15At := array.get(times15, pivot)
                    if okLo
                        lastLow15 := candL
                        lastLow15At := array.get(times15, pivot)

                // Comprobar rompimientos M15 correspondientes al MISMO CRT.
                // Se ejecuta ANTES del cierre H4: incluye el ultimo M15 de C2.
                int closed15At = m15LastEnd
                if c2Active and c2Dir != 0 and c2StrictSweep and not c2Double and closed15At > c2SweepEnd
                    bool eligible15 = c2Dir == 1 ? (not na(c2Pivot15) and m15Close > c2Pivot15 and (na(prev15Close) or prev15Close <= c2Pivot15)) : (not na(c2Pivot15) and m15Close < c2Pivot15 and (na(prev15Close) or prev15Close >= c2Pivot15))
                    if eligible15 and na(c2M15At)
                        c2M15At := closed15At
                if c3Active and closed15At > c3Start and closed15At > c3SweepEnd
                    bool eligible15 = c3Dir == 1 ? (not na(c3Pivot15) and m15Close > c3Pivot15 and (na(prev15Close) or prev15Close <= c3Pivot15)) : (not na(c3Pivot15) and m15Close < c3Pivot15 and (na(prev15Close) or prev15Close >= c3Pivot15))
                    if eligible15 and na(c3M15At)
                        c3M15At := closed15At
                        if lastC1 == c1Start3
                            lastM15 := closed15At
                prev15Close := m15Close

            m15Bucket := bucket15
            m15High := bH
            m15Low := bL
            m15Close := bC
            m15Count := 1
            m15LastEnd := barEnd
            m15Start := bucket15
        else if na(m15Bucket)
            m15Bucket := bucket15
            m15High := bH
            m15Low := bL
            m15Close := bC
            m15Count := 1
            m15LastEnd := barEnd
            m15Start := bucket15
        else
            if barT != m15LastEnd
                m15Count := -100 // M15 incompleta: no publicar MSS.
            else if m15Count >= 0
                m15Count += 1
            m15High := math.max(m15High, bH)
            m15Low := math.min(m15Low, bL)
            m15Close := bC
            m15LastEnd := barEnd

    // ---- CERRAR H4 ANTERIOR Y ABRIR NUEVA ----
    if not na(barT) and not na(bucketH4)
        bool nextH4 = not na(h4Bucket) and bucketH4 != h4Bucket
        if nextH4
            // C3 solo se invalida por el CIERRE de su H4, NO por cierre M5.
            // La mecha o un cierre intrabar M5 que atraviese el extremo de C2
            // no revocan el CRT cuando la C3 H4 finalmente cierra recuperada.
            if c3Active and c3Start == h4Bucket
                bool validC3Close = h4Continuous and h4FirstT == h4Bucket and h4LastEnd == bucketH4 and h4Count >= minBarsC2
                bool invalidAtC3Close = validC3Close and not na(c3Invalidation) and (c3Dir == 1 ? h4Close < c3Invalidation : h4Close > c3Invalidation)
                if invalidAtC3Close and lastC1 == c1Start3
                    lastConfirmedAt := na
                    lastM5 := na
                    lastM15 := na
                    lastModelKind := "NONE"
                    lastPatternValid := false
                    lastModelC1 := na
                c3Active := false
                c3ModelKind := "NONE"
                c3PatternValid := false
                c3ModelC1 := na
                c3C2Start := na

            // M3B.2 INTEGRADO: clasificar solo al cierre H4 de la manipulacion.
            // Una barrida M5 no equivale por si sola a un CRT ratificado.
            if array.size(insideCandidates) > 0
                for ic = 0 to array.size(insideCandidates) - 1
                    InsideCandidate cand = array.get(insideCandidates, ic)
                    bool closesSweepH4 = not na(cand.sweepH4Start) and cand.sweepH4Start == h4Bucket
                    if closesSweepH4 and (cand.state == "SWEPT" or cand.state == "AMBIGUOUS")
                        bool closedH4Complete = h4Continuous and h4FirstT == h4Bucket and h4LastEnd == bucketH4 and h4Count >= minBarsC2
                        if not closedH4Complete
                            cand.state := "DATA_GAP"
                            cand.isPatternValid := false
                        else if cand.state == "AMBIGUOUS" or (cand.sweptLow and cand.sweptHigh)
                            cand.state := "AMBIGUOUS"
                            cand.modelKind := "NONE"
                            cand.isPatternValid := false
                        else
                            bool returnC2 = h4Close >= cand.c2Low and h4Close <= cand.c2High
                            if returnC2
                                cand.state := "RATIFIED"
                                cand.modelKind := cand.sweepDir == 1 ? "INSIDE_RETURN_BULL" : "INSIDE_RETURN_BEAR"
                                cand.isPatternValid := true
                            else if h4Close < cand.c1Low or h4Close > cand.c1High
                                cand.state := "CONTINUATION"
                                cand.modelKind := h4Close < cand.c1Low ? "INSIDE_CONTINUATION_DOWN" : "INSIDE_CONTINUATION_UP"
                                cand.isPatternValid := false // Experimental, sin entradas
                            else
                                // Cierra dentro de C1 pero fuera de C2: sin clasificacion final.
                                cand.state := "UNRESOLVED"
                                cand.modelKind := "NONE"
                                cand.isPatternValid := false

            // ---- M3B.3B: EXPIRACION Y GAPS (DESPUES de clasificar 3B.2) ----
            // No confundir una pausa normal entre sesiones con un hueco M5
            // interno. Contamos cierres H4 verificados, no tiempo UTC / 4h.
            bool insideH4Closed = h4Continuous and h4FirstT == h4Bucket and h4Count >= minBarsC2 and (h4LastEnd == bucketH4 or h4LastEnd == h4Bucket + H4MS)
            if array.size(insideCandidates) > 0
                for ageI = 0 to array.size(insideCandidates) - 1
                    InsideCandidate ageCand = array.get(insideCandidates, ageI)
                    bool awaiting = ageCand.state == "SEEKING" or ageCand.state == "SWEPT"
                    bool eligibleH4 = h4Bucket >= ageCand.c2End
                    if awaiting and eligibleH4
                        if not h4Continuous
                            // Hay velas M5 en esta H4 pero falta continuidad interna.
                            ageCand.state := "DATA_GAP"
                            ageCand.isPatternValid := false
                        else if insideH4Closed
                            if ageCand.state == "SEEKING" and (h4Close < ageCand.c1Low or h4Close > ageCand.c1High)
                                ageCand.state := "INVALIDATED"
                                ageCand.isPatternValid := false
                            else
                                ageCand.h4Life += 1
                                if ageCand.h4Life >= maxH4Life
                                    ageCand.state := "EXPIRED"
                                    ageCand.isPatternValid := false
            // Los estados RATIFIED/CONTINUATION/AMBIGUOUS/UNRESOLVED son
            // terminales: no se reactivan ni consumen mas vida H4.

            // C2 pasa a C3 solo cuando la H4 de C2 CIERRA dentro de C1.
            bool c2Done = c2Active and c2Start == h4Bucket
            if c2Done
                bool inside = h4Close >= c1Low2 and h4Close <= c1High2
                // El modulo 1 permanece OBLIGATORIO para cualquier clasificacion.
                bool validC2Data = h4Continuous and h4FirstT == h4Bucket and h4LastEnd == bucketH4 and h4Count >= minBarsC2
                // M3B.1: C2 estrictamente interior a C1 al cierre H4 real.
                // No ratifica Inside, no altera C2-sweep, MSS ni dashboard.
                bool strictH4Inside = validC2Data and h4High < c1High2 and h4Low > c1Low2
                if strictH4Inside
                    string insideId = syminfo.tickerid + "|" + str.tostring(c1Start2) + "|INSIDE"
                    bool alreadyStored = false
                    int candidateCount = array.size(insideCandidates)
                    if candidateCount > 0
                        for j = 0 to candidateCount - 1
                            InsideCandidate existing = array.get(insideCandidates, j)
                            if existing.id == insideId
                                alreadyStored := true
                                break
                    if not alreadyStored
                        InsideCandidate cand = InsideCandidate.new(insideId, c1Start2, c1High2, c1Low2, h4High, h4Low, bucketH4, na, na, 0, false, false, "SEEKING", "NONE", false, 0)
                        // Una memoria llena descarta primero terminales sin
                        // ratificacion, despues SEEKING, SWEPT y finalmente
                        // el RATIFIED mas antiguo si todos son importantes.
                        if array.size(insideCandidates) >= 10
                            int dropIdx = na
                            for tier = 0 to 3
                                if na(dropIdx)
                                    for mi = 0 to array.size(insideCandidates) - 1
                                        string st = array.get(insideCandidates, mi).state
                                        bool discardable = tier == 0 ? (st == "EXPIRED" or st == "INVALIDATED" or st == "DATA_GAP" or st == "AMBIGUOUS" or st == "UNRESOLVED" or st == "CONTINUATION") : tier == 1 ? st == "SEEKING" : tier == 2 ? st == "SWEPT" : st == "RATIFIED"
                                        if discardable
                                            dropIdx := mi
                                            break
                            array.remove(insideCandidates, na(dropIdx) ? 0 : dropIdx)
                        array.push(insideCandidates, cand)
                bool strictSingleSweep = c2Dir != 0 and c2StrictSweep and not c2Double
                bool expansionBull = c2Dir == 1 and h4Close < c1Low2
                bool expansionBear = c2Dir == -1 and h4Close > c1High2
                string classified2C = "NONE"
                if validC2Data and strictSingleSweep
                    if inside and c2Returned
                        classified2C := "CRT_2C_RETURN"
                    else if expansionBull or expansionBear
                        classified2C := "CRT_2C_EXPANSION"  // Solo diagnostico, NUNCA ratificado
                last2CClassification := classified2C
                last2CClassificationAt := bucketH4
                bool validRange = classified2C == "CRT_2C_RETURN"
                bool exactHour3 = hour(bucketH4, TZ) == ((hour(c1Start2, TZ) + 8) % 24)
                if validRange and exactHour3
                    // Registrar el CRT al CIERRE H4 de C2. M5/M15 se
                    // conservan solo si ya se confirmaron en esa C2.
                    lastC1 := c1Start2
                    lastDir := c2Dir
                    lastConfirmedAt := bucketH4
                    lastHi := c1High2
                    lastLo := c1Low2
                    lastSweepAt := c2SweepEnd
                    lastM5 := c2M5At
                    lastM15 := c2M15At
                    lastM5Level := c2Pivot5
                    lastM15Level := c2Pivot15
                    lastM5PivotAt := c2Pivot5At
                    lastM15PivotAt := c2Pivot15At
                    lastModelKind := classified2C
                    lastPatternValid := true
                    lastModelC1 := c1Start2
                    lastC2Start := c2Start
                    c3ModelKind := classified2C
                    c3PatternValid := true
                    c3ModelC1 := c1Start2
                    c3C2Start := c2Start
                    c3Active := true
                    c3Start := bucketH4
                    c1Start3 := c1Start2
                    c1High3 := c1High2
                    c1Low3 := c1Low2
                    c3Dir := c2Dir
                    c3SweepEnd := c2SweepEnd
                    c3Pivot5 := c2Pivot5
                    c3Pivot15 := c2Pivot15
                    c3Pivot5At := c2Pivot5At
                    c3Pivot15At := c2Pivot15At
                    c3M5At := c2M5At
                    c3M15At := c2M15At
                    c3Invalidation := c2Dir == 1 ? h4Low : h4High
                c2Active := false

            // La H4 que acaba de cerrar puede ser C1 del SIGUIENTE ciclo,
            // incluso si la H4 nueva ya es C3 de un ciclo anterior.
            c2Active := false
            c2Dir := 0
            c2StrictSweep := false
            c2Double := false
            c2Returned := false
            c2SweepEnd := na
            c2M5At := na
            c2M15At := na
            c2Pivot5 := na
