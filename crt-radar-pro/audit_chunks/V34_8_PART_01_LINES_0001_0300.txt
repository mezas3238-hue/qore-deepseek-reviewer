f_index() =>
    string name = str.upper(syminfo.ticker)
    string csv = "," + str.upper(str.replace_all(idxList, " ", "")) + ","
    bool listed = str.contains(csv, "," + name + ",")
    bool typical = str.contains(name, "NAS100") or str.contains(name,"US100") or str.contains(name,"US500") or str.contains(name,"SPX500") or str.contains(name,"US30") or str.contains(name,"US2000") or str.contains(name,"DE40") or str.contains(name,"GER40") or str.contains(name,"UK100") or str.contains(name,"JP225") or str.contains(name,"HK50")
    syminfo.type == "index" or listed or typical

f_allowed(int hh, bool indexFamily) =>
    // Cada H4 de la sesion NY es una posible C1, incluso cuando otra es C2/C3.
    bool exact = indexFamily ? (hh == 2 or hh == 6 or hh == 22) : (hh == 1 or hh == 5 or hh == 21)
    bool anyH4 = indexFamily ? (hh == 2 or hh == 6 or hh == 10 or hh == 14 or hh == 18 or hh == 22) : (hh == 1 or hh == 5 or hh == 9 or hh == 13 or hh == 17 or hh == 21)
    scheduleMode == "Todas las H4 NY" ? anyH4 : exact

// En M5 reconstruimos tambien las velas M15 completas, de modo que se usa

f_engine() =>
    bool isIndex = f_index()
    int barT = time[1]
    int barEnd = time_close[1]
    float bH = high[1]
    float bL = low[1]
    float bC = close[1]
    float prevClose = close[2]

    // M4A: PDH/PDL calculados en la misma secuencia M5 cerrada.
    // prev_valid solo significa que la ULTIMA sesion finalizada paso control
    // de calidad; no se reutiliza silenciosamente un nivel de hace dias.
    var float curr_H = na
    var float curr_L = na
    var int curr_sess_id = na
    var int curr_m5_count = 0
    var int curr_firstT = na
    var int curr_lastEnd = na
    var bool curr_continuous = true
    var float prev_PDH = na
    var float prev_PDL = na
    var bool prev_valid = false
    var int prev_timestamp = na
    var int prev_sess_id = na
    // M4B: el ID del nivel es su fecha de confirmacion 4A.
    // No se consulta curr_sess_id[1] ni se reevalua una vela incompleta.
    var int liquidityRefAt = na
    var bool pdhTaken = false
    var bool pdlTaken = false
    var int pdhSweepAt = na
    var int pdlSweepAt = na
    var int pdhSweepStage = 0
    var int pdlSweepStage = 0
    var int pdhSweepC1 = na
    var int pdlSweepC1 = na

    // ---- M4A PDH / PDL: SIN solicitudes adicionales ----
    // Time() se consulta con [1] igual que barT/bH/bL: M5 CERRADA.
    bool pdhRth = isIndex and not useExtIdx
    string pdhSession = pdhRth ? "0930-1600:23456" : "1700-1700:23456"
    bool pdhInSession = not na(time("5", pdhSession, TZ)[1])
    if not na(barT) and not na(barEnd) and not na(bH) and not na(bL)
        // Identidad calendario local del INICIO de sesion; segura en DST.
        // Mediodia menos 24h identifica el DIA ANTERIOR incluso si DST
        // cambio el desfase UTC una hora durante la madrugada.
        int pdhNoon = timestamp(TZ, year(barT, TZ), month(barT, TZ), dayofmonth(barT, TZ), 12, 0)
        int pdhAnchor = pdhRth or hour(barT, TZ) >= 17 ? pdhNoon : pdhNoon - 86400000
        int pdhKey = year(pdhAnchor, TZ) * 10000 + month(pdhAnchor, TZ) * 100 + dayofmonth(pdhAnchor, TZ)
        if pdhInSession
            if na(curr_sess_id) or pdhKey != curr_sess_id
                // Si un dia termino sin vela de cierre, NO publicar parcial.
                if not na(curr_sess_id)
                    prev_valid := false
                curr_sess_id := pdhKey
                curr_H := bH
                curr_L := bL
                curr_m5_count := 1
                curr_firstT := barT
                curr_lastEnd := barEnd
                curr_continuous := true
            else
                curr_continuous := curr_continuous and barT == curr_lastEnd
                curr_H := math.max(curr_H, bH)
                curr_L := math.min(curr_L, bL)
                curr_m5_count += 1
                curr_lastEnd := barEnd
            // Con la ultima M5 (15:55-16 o 16:55-17) se cierra
            // la sesion, aunque el proveedor no tenga velas posteriores.
            int endHour = pdhRth ? 16 : 17
            bool sessClosed = hour(barEnd, TZ) == endHour and minute(barEnd, TZ) == 0
            if sessClosed
                int firstHour = pdhRth ? 9 : 17
                int firstMinute = minute(curr_firstT, TZ)
                bool startedOnTime = hour(curr_firstT, TZ) == firstHour and (pdhRth ? firstMinute == 30 : firstMinute <= 5)
                bool complete = curr_continuous and startedOnTime and curr_m5_count >= (pdhRth ? 78 : 280)
                prev_valid := complete
                if complete
                    prev_PDH := curr_H
                    prev_PDL := curr_L
                    prev_timestamp := barEnd
                    prev_sess_id := curr_sess_id
                // Valores historicos se conservan con prev_valid=false.
                curr_sess_id := na
                curr_H := na
                curr_L := na
                curr_m5_count := 0
                curr_firstT := na
                curr_lastEnd := na
                curr_continuous := true
        else if not na(curr_sess_id)
            // Fin de sesion sin M5 de cierre: no falsificar PDH/PDL.
            prev_valid := false
            curr_sess_id := na
            curr_m5_count := 0
            curr_H := na
            curr_L := na
            curr_firstT := na
            curr_lastEnd := na
            curr_continuous := true

    // Pivot M5: calculo en la barra cerrada [1]; nunca es una barra futura.
    float pH5 = ta.pivothigh(high, left5, right5)[1]
    float pL5 = ta.pivotlow(low, left5, right5)[1]
    var float lastHigh5 = na
    var float lastLow5 = na
    var int lastHigh5At = na
    var int lastLow5At = na

    // Acumulador M15 sobre barras M5 completas.
    var int m15Bucket = na
    var float m15High = na
    var float m15Low = na
    var float m15Close = na
    var int m15Count = 0
    var int m15LastEnd = na
    var int m15Start = na
    var array<float> highs15 = array.new_float()
    var array<float> lows15 = array.new_float()
    var array<int> times15 = array.new_int()
    var float lastHigh15 = na
    var float lastLow15 = na
    var int lastHigh15At = na
    var int lastLow15At = na
    var float prev15Close = na

    // H4 reconstruida desde datos de M5 del mismo proveedor.
    var int h4Bucket = na
    var float h4High = na
    var float h4Low = na
    var float h4Close = na
    var int h4Count = 0
    var int h4FirstT = na
    var int h4LastEnd = na
    // Modulo 1 aprobado: rechaza H4 con huecos M5 internos.
    var bool h4Continuous = true

    // C2 actual: puede coexistir con la C3 del CRT anterior.
    // M3B.1: memoria independiente por contexto de simbolo (maximo 10).
    var array<InsideCandidate> insideCandidates = array.new<InsideCandidate>()

    var bool c2Active = false
    var int c2Start = na
    var int c1Start2 = na
    var float c1High2 = na
    var float c1Low2 = na
    var int c2Dir = 0
    var bool c2StrictSweep = false
    var bool c2Double = false
    var bool c2Returned = false
    var int c2SweepEnd = na
    var float c2Pivot5 = na
    var float c2Pivot15 = na
    var int c2Pivot5At = na
    var int c2Pivot15At = na
    var int c2M5At = na
    var int c2M15At = na
