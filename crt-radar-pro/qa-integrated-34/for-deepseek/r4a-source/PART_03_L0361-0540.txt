// Funciones Pine deben definirse en el ambito global.
f_add_context_fvg(array<FvgContext> store, string tf, int dir, float top, float bottom, int formedAt, bool endSessionOnFormation) =>
    string k = syminfo.tickerid + "|" + tf + "|" + str.tostring(formedAt) + "|" + str.tostring(dir)
    bool exists = false
    if array.size(store) > 0
        for z = 0 to array.size(store) - 1
            if array.get(store, z).id == k
                exists := true
                break
    if not exists
        array.push(store, FvgContext.new(k, tf, dir, top, bottom, formedAt, "INTACTO", na, na, 0.0, false, formedAt, na, endSessionOnFormation))
        if array.size(store) > 10
            array.shift(store)
    not exists

// M5B CANDIDATO NO INTEGRADO: actualizar un FVG con la M5 cerrada.
// Un corte de observabilidad (incluidos cierres de mercado no clasificados)
// congela el estado, NO inventa mitigacion en barras que faltan.
f_update_context_fvg(FvgContext f, int start5, int end5, float lo5, float hi5, float close5, bool firstSession5, bool lastSession5) =>
    bool eligible = not na(start5) and not na(end5) and start5 >= f.formedAt and not na(f.lastEvaluatedEnd) and end5 > f.lastEvaluatedEnd
    if eligible and f.state != "INVALIDADO" and f.state != "DATOS_INCOMPLETOS"
        // Solo el par ultimo-barra-sesion / primer-barra-sesion justifica
        // tratar la pausa como no-trading. NO saltar huecos por dayofweek.
        bool knownClosedSession = start5 > f.lastEvaluatedEnd and f.lastSessionBar and firstSession5
        bool discontinuity = (start5 != f.lastEvaluatedEnd and not knownClosedSession) or end5 - start5 != MIN5 or na(lo5) or na(hi5) or na(close5)
        if discontinuity
            f.dataGap := true
            f.gapObservedAt := start5
            f.state := "DATOS_INCOMPLETOS"
        else
            float width = f.top - f.bottom
            if width > 0.0 and (f.dir == 1 or f.dir == -1)
                bool touched = f.dir == 1 ? lo5 <= f.top : hi5 >= f.bottom
                if touched and na(f.firstTouchAt)
                    f.firstTouchAt := end5
                if touched
                    float rawPen = f.dir == 1 ? (f.top - lo5) / width : (hi5 - f.bottom) / width
                    f.maxPenetration := math.max(f.maxPenetration, math.min(1.0, math.max(0.0, rawPen)))
                bool broken = f.dir == 1 ? close5 < f.bottom : close5 > f.top
                if broken
                    f.state := "INVALIDADO"
                else if f.maxPenetration >= 1.0
                    f.state := "LLENADO_TOTAL"
                else if not na(f.firstTouchAt)
                    // Contacto justo en el borde conserva penetracion 0%.
                    f.state := "PARCIAL"
                else
                    f.state := "INTACTO"
            else
                f.dataGap := true
                f.gapObservedAt := start5
                f.state := "DATOS_INCOMPLETOS"
        f.lastEvaluatedEnd := end5
        f.lastSessionBar := lastSession5
    // Modifica los campos del UDT por referencia; sin reasignar f.
    f.state

// QA: -1=no existe, 0 intacto, 1 parcial, 2 lleno, 3 invalidado, 4 datos incompletos.
f_fvg_code(string st) =>
    st == "INTACTO" ? 0 : st == "PARCIAL" ? 1 : st == "LLENADO_TOTAL" ? 2 : st == "INVALIDADO" ? 3 : st == "DATOS_INCOMPLETOS" ? 4 : -2

// Every M15 candidate must close on this provider's own native phase.
// 0..4 are exact minute offsets, not M5->M15 substitutions.
f_m15_phase_ok(int closedAt, int sourcePhaseMin) =>
    not na(closedAt) and sourcePhaseMin >= 0 and sourcePhaseMin <= 4 and closedAt % MIN15 == sourcePhaseMin * 60000
