
// V34.11e: deterministic index H4 alignment on NY 02/06/10/14/18/22.
// 00:00-01:59 NY belongs to yesterday's 22:00 H4.
// A nonexistent NY 02:00 DST spring anchor cannot be made a valid C1
// by relaxing the 48-M5-contiguous candle gate.
f_index_crt_h4_ny(int t) =>
    int hr = hour(t, TZ)
    int bucketHr = hr >= 22 ? 22 : hr >= 18 ? 18 : hr >= 14 ? 14 : hr >= 10 ? 10 : hr >= 6 ? 6 : hr >= 2 ? 2 : 22
    int dateRef = hr < 2 ? t - 12 * 3600000 : t
    timestamp(TZ, year(dateRef, TZ), month(dateRef, TZ), dayofmonth(dateRef, TZ), bucketHr, 0)
