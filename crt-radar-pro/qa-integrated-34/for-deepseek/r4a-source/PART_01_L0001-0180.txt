f_index() =>
    string name = str.upper(syminfo.ticker)
    string csv = "," + str.upper(str.replace_all(idxList, " ", "")) + ","
    bool listed = str.contains(csv, "," + name + ",")
    bool typical = str.contains(name, "USTEC") or str.contains(name, "NAS100") or str.contains(name,"US100") or str.contains(name,"US500") or str.contains(name,"SPX500") or str.contains(name,"US30") or str.contains(name,"US2000") or str.contains(name,"DE40") or str.contains(name,"GER40") or str.contains(name,"UK100") or str.contains(name,"JP225") or str.contains(name,"HK50")
    syminfo.type == "index" or listed or typical

f_allowed(int hh, bool indexFamily) =>
    // Ancla horaria NY NO equivale a operar solo la SESION de Nueva York.
    // Todos los 6 bloques H4 del ciclo de 24h son elegibles por defecto,
    // incluyendo Asia, Londres y NY; el motor siempre valida 48 M5
    // contiguas, CRH/CRL y C2 cerrada por separado.
    // Solo 6/6 H4, incluye Asia/Londres/NY. Sin selector de 3 ventanas.
    indexFamily ? (hh == 2 or hh == 6 or hh == 10 or hh == 14 or hh == 18 or hh == 22) : (hh == 1 or hh == 5 or hh == 9 or hh == 13 or hh == 17 or hh == 21)

// BTC: construir H4 desde la HORA LOCAL NY de CADA M5, nunca
// desde time("240", "0100-0100...") que en la salida DST puede
// anclar a 00/04/08 y saltarse C1 01/05/09. Anclas validas de
// cualquier dia: 01/05/09/13/17/21. De 00:00 a 00:59 corresponde
// al bloque 21:00 del DIA NY ANTERIOR.
// En el retroceso DST, 01:00 ocurre dos veces; las M5 del bloque
// excederan 48 y las puertas de C1/C2/C3 bloquearan el CRT.
f_btc_crt_h4_ny(int t) =>
    int hr = hour(t, TZ)
    int bucketHr = hr >= 21 ? 21 : hr >= 17 ? 17 : hr >= 13 ? 13 : hr >= 9 ? 9 : hr >= 5 ? 5 : hr >= 1 ? 1 : 21
    // Solo 00 NY pertenece al dia previo; -12 horas garantiza
    // fecha NY anterior aun en transiciones DST.
    int dateRef = hr == 0 ? t - 12 * 3600000 : t
    timestamp(TZ, year(dateRef, TZ), month(dateRef, TZ), dayofmonth(dateRef, TZ), bucketHr, 0)

// V34.11f: NON-BTC markets have a dedicated NY anchoring function.
// Forex and metals are checked independently from BTCUSD; the latter
// keeps its original clock and validation. Trading session coverage for
// XAU/XAG is NOT assumed equivalent to liquid Forex.
f_forex_metal_crt_h4_ny(int t) =>
    int hr = hour(t, TZ)
    int bucketHr = hr >= 21 ? 21 : hr >= 17 ? 17 : hr >= 13 ? 13 : hr >= 9 ? 9 : hr >= 5 ? 5 : hr >= 1 ? 1 : 21
    int dateRef = hr == 0 ? t - 12 * 3600000 : t
    timestamp(TZ, year(dateRef, TZ), month(dateRef, TZ), dayofmonth(dateRef, TZ), bucketHr, 0)
