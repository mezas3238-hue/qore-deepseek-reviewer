// Funciones Pine deben definirse en el ambito global.
f_add_context_fvg(array<FvgContext> store, string tf, int dir, float top, float bottom, int formedAt, bool endSessionOnFormation) =>
    string k = syminfo.tickerid + "|" + tf + "|" + str.tostring(formedAt) + "|" + str.tostring(dir)
    bool exists = false
    if array.size(store) > 0
        for z = 0 to array.size(store) - 1
            if array.get(store, z).id == k
                exists := true
                break
    if not exists
        array.push(store, FvgContext.new(k, tf, dir, top, bottom, formedAt, "INTACTO", na, na, 0.0, false, formedAt, na, endSessionOnFormation))
        if array.size(store) > 10
            array.shift(store)
    not exists

// M5B CANDIDATO NO INTEGRADO: actualizar un FVG con la M5 cerrada.
// Un corte de observabilidad (incluidos cierres de mercado no clasificados)
// congela el estado, NO inventa mitigacion en barras que faltan.
f_update_context_fvg(FvgContext f, int start5, int end5, float lo5, float hi5, float close5, bool firstSession5, bool lastSession5) =>
    bool eligible = not na(start5) and not na(end5) and start5 >= f.formedAt and not na(f.lastEvaluatedEnd) and end5 > f.lastEvaluatedEnd
    if eligible and f.state != "INVALIDADO" and f.state != "DATOS_INCOMPLETOS"
        // Solo el par ultimo-barra-sesion / primer-barra-sesion justifica
        // tratar la pausa como no-trading. NO saltar huecos por dayofweek.
        bool knownClosedSession = start5 > f.lastEvaluatedEnd and f.lastSessionBar and firstSession5
        bool discontinuity = (start5 != f.lastEvaluatedEnd and not knownClosedSession) or end5 - start5 != MIN5 or na(lo5) or na(hi5) or na(close5)
        if discontinuity
            f.dataGap := true
            f.gapObservedAt := start5
            f.state := "DATOS_INCOMPLETOS"
        else
            float width = f.top - f.bottom
            if width > 0.0 and (f.dir == 1 or f.dir == -1)
                bool touched = f.dir == 1 ? lo5 <= f.top : hi5 >= f.bottom
                if touched and na(f.firstTouchAt)
                    f.firstTouchAt := end5
                if touched
                    float rawPen = f.dir == 1 ? (f.top - lo5) / width : (hi5 - f.bottom) / width
                    f.maxPenetration := math.max(f.maxPenetration, math.min(1.0, math.max(0.0, rawPen)))
                bool broken = f.dir == 1 ? close5 < f.bottom : close5 > f.top
                if broken
                    f.state := "INVALIDADO"
                else if f.maxPenetration >= 1.0
                    f.state := "LLENADO_TOTAL"
                else if not na(f.firstTouchAt)
                    // Contacto justo en el borde conserva penetracion 0%.
                    f.state := "PARCIAL"
                else
                    f.state := "INTACTO"
            else
                f.dataGap := true
                f.gapObservedAt := start5
                f.state := "DATOS_INCOMPLETOS"
        f.lastEvaluatedEnd := end5
        f.lastSessionBar := lastSession5
    // Modifica los campos del UDT por referencia; sin reasignar f.
    f.state

// QA: -1=no existe, 0 intacto, 1 parcial, 2 lleno, 3 invalidado, 4 datos incompletos.
f_fvg_code(string st) =>
    st == "INTACTO" ? 0 : st == "PARCIAL" ? 1 : st == "LLENADO_TOTAL" ? 2 : st == "INVALIDADO" ? 3 : st == "DATOS_INCOMPLETOS" ? 4 : -2

f_engine() =>
    bool isIndex = f_index()
    bool isBTC = str.upper(syminfo.ticker) == "BTCUSD"
    int barT = time[1]
    int barEnd = time_close[1]
    float bH = high[1]
    float bL = low[1]
    float bC = close[1]
    float prevClose = close[2]
    // Datos de sesiones del MISMO ticker M5 dentro del request.security.
    bool firstSession5 = session.isfirstbar[1]
    bool lastSession5 = session.islastbar[1]
    bool previousSessionLast5 = session.islastbar[2]
    int previousBarEnd = time_close[2]
    // QA: la M5 previa debe finalizar exactamente donde inicia la actual.
    // Pine [2] recorre barras, no garantiza ausencia de huecos temporales.
    bool previousM5Continuous = not na(barT) and not na(previousBarEnd) and barT == previousBarEnd

    // M4A: PDH/PDL calculados en la misma secuencia M5 cerrada.
    // prev_valid solo significa que la ULTIMA sesion finalizada paso control
    // de calidad; no se reutiliza silenciosamente un nivel de hace dias.
    var float curr_H = na
    var float curr_L = na
    var int curr_sess_id = na
    var int curr_m5_count = 0
    var int curr_firstT = na
    var int curr_lastEnd = na
    var bool curr_continuous = true
    var float prev_PDH = na
    var float prev_PDL = na
    var bool prev_valid = false
    var int prev_timestamp = na
    var int prev_sess_id = na
    // M4B: el ID del nivel es su fecha de confirmacion 4A.
    // No se consulta curr_sess_id[1] ni se reevalua una vela incompleta.
    var int liquidityRefAt = na
    var bool pdhTaken = false
    var bool pdlTaken = false
    var int pdhSweepAt = na
    var int pdlSweepAt = na
    var int pdhSweepStage = 0
    var int pdlSweepStage = 0
    var int pdhSweepC1 = na
    var int pdlSweepC1 = na

    // ---- M4A PDH / PDL: SIN solicitudes adicionales ----
    // Time() se consulta con [1] igual que barT/bH/bL: M5 CERRADA.
    bool pdhRth = isIndex and not useExtIdx
    string pdhSession = pdhRth ? "0930-1600:23456" : "1700-1700:23456"
    bool pdhInSession = not na(time("5", pdhSession, TZ)[1])
    if not na(barT) and not na(barEnd) and not na(bH) and not na(bL)
        // Identidad calendario local del INICIO de sesion; segura en DST.
        // Mediodia menos 24h identifica el DIA ANTERIOR incluso si DST
        // cambio el desfase UTC una hora durante la madrugada.
        int pdhNoon = timestamp(TZ, year(barT, TZ), month(barT, TZ), dayofmonth(barT, TZ), 12, 0)
        int pdhAnchor = pdhRth or hour(barT, TZ) >= 17 ? pdhNoon : pdhNoon - 86400000
        int pdhKey = year(pdhAnchor, TZ) * 10000 + month(pdhAnchor, TZ) * 100 + dayofmonth(pdhAnchor, TZ)
        if pdhInSession
            if na(curr_sess_id) or pdhKey != curr_sess_id
                // Si un dia termino sin vela de cierre, NO publicar parcial.
                if not na(curr_sess_id)
                    prev_valid := false
                curr_sess_id := pdhKey
                curr_H := bH
                curr_L := bL
                curr_m5_count := 1
                curr_firstT := barT
                curr_lastEnd := barEnd
                curr_continuous := true
            else
                curr_continuous := curr_continuous and barT == curr_lastEnd
                curr_H := math.max(curr_H, bH)
                curr_L := math.min(curr_L, bL)
                curr_m5_count += 1
                curr_lastEnd := barEnd
            // Con la ultima M5 (15:55-16 o 16:55-17) se cierra
            // la sesion, aunque el proveedor no tenga velas posteriores.
            int endHour = pdhRth ? 16 : 17
            bool sessClosed = hour(barEnd, TZ) == endHour and minute(barEnd, TZ) == 0
            if sessClosed
                int firstHour = pdhRth ? 9 : 17
                int firstMinute = minute(curr_firstT, TZ)
                bool startedOnTime = hour(curr_firstT, TZ) == firstHour and (pdhRth ? firstMinute == 30 : firstMinute <= 5)
                bool complete = curr_continuous and startedOnTime and curr_m5_count >= (pdhRth ? 78 : 280)
                prev_valid := complete
                if complete
                    prev_PDH := curr_H
                    prev_PDL := curr_L
                    prev_timestamp := barEnd
                    prev_sess_id := curr_sess_id
                // Valores historicos se conservan con prev_valid=false.
                curr_sess_id := na
                curr_H := na
                curr_L := na
                curr_m5_count := 0
                curr_firstT := na
                curr_lastEnd := na
                curr_continuous := true
        else if not na(curr_sess_id)
            // Fin de sesion sin M5 de cierre: no falsificar PDH/PDL.
            prev_valid := false
            curr_sess_id := na
            curr_m5_count := 0
            curr_H := na
            curr_L := na
            curr_firstT := na
            curr_lastEnd := na
            curr_continuous := true

    // Pivot M5: calculo en la barra cerrada [1]; nunca es una barra futura.

    float pL5 = ta.pivotlow(low, left5, right5)[1]
    var float lastHigh5 = na

    var int lastHigh5At = na

    var int m15Count = 0
    var int m15LastEnd = na
    var int m15Start = na

    var float prev15Close = na
    // QA: solo comparar cierres M15 consecutivos completos (no saltar sesiones).
    var int prev15End = na

    // H4 reconstruida desde datos de M5 del mismo proveedor.
    var int h4Bucket = na
    var float h4High = na
    var float h4Low = na
    var float h4Close = na
    var int h4Count = 0
    var int h4FirstT = na
    var int h4LastEnd = na
    // Modulo 1 aprobado: rechaza H4 con huecos M5 internos.

    // M5A: Reconstruccion H1 desde M5 cerradas y memoria de 3 barras TF.
    var int h1Bucket = na
    var int h1FirstT = na
    var int h1LastEnd = na
    var float h1High = na
    var float h1Low = na
    var int h1Count = 0
    var bool h1Continuous = true
    var FvgBar h1Prev = na

    var FvgBar h4Prev2 = na
    var array<FvgContext> fvgsH1 = array.new<FvgContext>()
    var array<FvgContext> fvgsH4 = array.new<FvgContext>()

    // C2 actual: puede coexistir con la C3 del CRT anterior.
    // M3B.1: memoria independiente por contexto de simbolo (maximo 10).
    var array<InsideCandidate> insideCandidates = array.new<InsideCandidate>()

    var bool c2Active = false
    var int c2Start = na
    var int c1Start2 = na
    var float c1High2 = na
    var float c1Low2 = na
    var int c2Dir = 0
    var bool c2StrictSweep = false
    var bool c2Double = false
    var bool c2Returned = false
    var int c2SweepEnd = na
    var float c2Pivot5 = na
    var float c2Pivot15 = na
    var int c2Pivot5At = na
    var int c2Pivot15At = na
    var int c2M5At = na
    var int c2M15At = na

    // C3 ratificada: los MSS de C2 se heredan, los faltantes se esperan en C3.
    var bool c3Active = false
    var int c3Start = na
    var int c1Start3 = na
    var float c1High3 = na
    var float c1Low3 = na
    var int c3Dir = 0
    var int c3SweepEnd = na
    var float c3Pivot5 = na
    var float c3Pivot15 = na
    var int c3Pivot5At = na
    var int c3Pivot15At = na
    var int c3M5At = na
    var int c3M15At = na
    var float c3Invalidation = na
    // Metadatos del CRT ratificado que se encuentra en C3.
    var string c3ModelKind = "NONE"
    var bool c3PatternValid = false
    var int c3ModelC1 = na
    var int c3C2Start = na // Solo metadato para asociar PDH/PDL, sin alterar MSS.

    // Clasificacion diagnostica del ultimo cierre C2, aun si no es operativo.
    var string last2CClassification = "NONE"
    var int last2CClassificationAt = na

    // Memoria controlada del ULTIMO CRT realmente ratificado, NO de
    // cualquier MSS aislado. Los flags se copian desde C2/C3 y no se
    // inventan desde eventos antiguos no vinculados al ciclo.
    var int lastC1 = na
    var int lastDir = 0
    var int lastConfirmedAt = na
    var float lastHi = na
    var float lastLo = na
    var int lastSweepAt = na
    var int lastM5 = na
    var int lastM15 = na
    var float lastM5Level = na
    var float lastM15Level = na
    var int lastM5PivotAt = na
    var int lastM15PivotAt = na
    var string lastModelKind = "NONE"
    var bool lastPatternValid = false
    var int lastModelC1 = na
    var int lastC2Start = na // Memoria del mismo CRT ratificado.

    // Procesamos la ULTIMA M5 cerrada, no la M5 en formacion.
    // f_engine() debe evaluarse incondicionalmente en cada barra M5.
    int bucketH1 = time("60")[1]
    int bucket15 = time("15")[1]
    int bucketH4 = isIndex ? time("240", "0200-0200:1234567", TZ)[1] : time("240", "0100-0100:1234567", TZ)[1]

    // QA: al aparecer una nueva M5 despues de un hueco, descartar
    // acumuladores y swings M15 de la cadena anterior. No permitir que
    // pivotes confirmados con barras discontinuas crucen a un nuevo CRT.
    if not previousM5Continuous and not na(barT)
        m15Bucket := na
        m15Count := 0
        m15LastEnd := na
        m15Start := na
        prev15Close := na
        prev15End := na
        array.clear(highs15)
        array.clear(lows15)
        array.clear(times15)
        lastHigh15 := na
        lastLow15 := na
        lastHigh15At := na
        lastLow15At := na

    // ---- CERRAR M15 ANTERIOR ----
    if not na(barT) and not na(bucket15)
        bool next15 = not na(m15Bucket) and bucket15 != m15Bucket
        if next15
            bool fullM15 = m15Count == 3 and m15LastEnd == bucket15 and m15Start + MIN15 == m15LastEnd
            if fullM15
                array.push(highs15, m15High)
                array.push(lows15, m15Low)
                array.push(times15, m15Start)
                if array.size(highs15) > 48
                    array.shift(highs15)
                    array.shift(lows15)
                    array.shift(times15)
                int n = array.size(highs15)
                int pivot = n - 1 - right15
                if pivot >= left15 and pivot >= 0
                    float candH = array.get(highs15, pivot)
                    float candL = array.get(lows15, pivot)
                    bool okHi = true
                    bool okLo = true
                    for j = -left15 to right15
                        if j != 0
                            okHi := okHi and candH > array.get(highs15, pivot + j)
                            okLo := okLo and candL < array.get(lows15, pivot + j)
                    if okHi
                        lastHigh15 := candH
                        lastHigh15At := array.get(times15, pivot)
                    if okLo
                        lastLow15 := candL
                        lastLow15At := array.get(times15, pivot)

                // QA: MSS M15 requiere cierre anterior INMEDIATO y completo.
                // Tras huecos, inicios de feed o pausas no se inventa cross.
                bool prev15Continuous = not na(prev15End) and prev15End == m15Start and not na(prev15Close)
                // Comprobar rompimientos M15 correspondientes al MISMO CRT.
                // Se ejecuta ANTES del cierre H4: incluye el ultimo M15 de C2.
                int closed15At = m15LastEnd
                if c2Active and c2Dir != 0 and c2StrictSweep and not c2Double and closed15At > c2SweepEnd
                    bool eligible15 = prev15Continuous and (c2Dir == 1 ? (not na(c2Pivot15) and m15Close > c2Pivot15 and prev15Close <= c2Pivot15) : (not na(c2Pivot15) and m15Close < c2Pivot15 and prev15Close >= c2Pivot15))
                    if eligible15 and na(c2M15At)
                        c2M15At := closed15At
                if c3Active and closed15At > c3Start and closed15At > c3SweepEnd
                    bool eligible15 = prev15Continuous and (c3Dir == 1 ? (not na(c3Pivot15) and m15Close > c3Pivot15 and prev15Close <= c3Pivot15) : (not na(c3Pivot15) and m15Close < c3Pivot15 and prev15Close >= c3Pivot15))
                    if eligible15 and na(c3M15At)
                        c3M15At := closed15At
                        if lastC1 == c1Start3
                            lastM15 := closed15At
                prev15Close := m15Close
                prev15End := m15LastEnd
            else
                // Una M15 omitida o incompleta invalida pivotes construidos
                // con muestras discontinuas, y el cierre previo del MSS.
                prev15Close := na
                prev15End := na
                array.clear(highs15)
                array.clear(lows15)
                array.clear(times15)
                lastHigh15 := na
                lastLow15 := na
                lastHigh15At := na
                lastLow15At := na

            m15Bucket := bucket15
            m15High := bH
            m15Low := bL
            m15Close := bC
            m15Count := 1
            m15LastEnd := barEnd
            m15Start := bucket15
        else if na(m15Bucket)
            m15Bucket := bucket15
            m15High := bH
            m15Low := bL
            m15Close := bC
            m15Count := 1
            m15LastEnd := barEnd
            m15Start := bucket15
        else
            if barT != m15LastEnd
                m15Count := -100 // M15 incompleta: no publicar MSS.
                // No conservar swings ni el cierre de un M15 sin continuidad.
                prev15Close := na
                prev15End := na
                array.clear(highs15)
                array.clear(lows15)
                array.clear(times15)
                lastHigh15 := na
                lastLow15 := na
                lastHigh15At := na
                lastLow15At := na
            else if m15Count >= 0
                m15Count += 1
            m15High := math.max(m15High, bH)
            m15Low := math.min(m15Low, bL)
            m15Close := bC
            m15LastEnd := barEnd

    // ---- M5A: CIERRE H1, FVG SIN LOOKAHEAD ----
    if not na(barT) and not na(bucketH1)
        bool nextH1 = not na(h1Bucket) and bucketH1 != h1Bucket
        if nextH1
            bool fullH1 = h1Continuous and h1Count == 12 and h1FirstT == h1Bucket and h1LastEnd == h1Bucket + 3600000
            if fullH1 and not na(h1Prev) and not na(h1Prev2)
                bool chainH1 = h1Prev.valid and h1Prev2.valid and h1Prev2.endAt == h1Prev.startAt and h1Prev.endAt == h1Bucket
                if chainH1
                    if h1Low > h1Prev2.hi
                        f_add_context_fvg(fvgsH1, "H1", 1, h1Low, h1Prev2.hi, h1LastEnd, previousBarEnd == h1LastEnd and previousSessionLast5)
                    else if h1High < h1Prev2.lo
                        f_add_context_fvg(fvgsH1, "H1", -1, h1Prev2.lo, h1High, h1LastEnd, previousBarEnd == h1LastEnd and previousSessionLast5)
            h1Prev2 := h1Prev
            h1Prev := FvgBar.new(h1Bucket, h1LastEnd, h1High, h1Low, fullH1)
            h1Bucket := bucketH1
            h1FirstT := barT
            h1LastEnd := barEnd
            h1High := bH
            h1Low := bL
            h1Count := 1
            h1Continuous := true
        else if na(h1Bucket)
            h1Bucket := bucketH1
            h1FirstT := barT
            h1LastEnd := barEnd
            h1High := bH
            h1Low := bL
            h1Count := 1
            h1Continuous := true
        else
            h1Continuous := h1Continuous and barT == h1LastEnd
            h1High := math.max(h1High, bH)
            h1Low := math.min(h1Low, bL)
            h1Count += 1
            h1LastEnd := barEnd

    // ---- CERRAR H4 ANTERIOR Y ABRIR NUEVA ----
    if not na(barT) and not na(bucketH4)
        bool nextH4 = not na(h4Bucket) and bucketH4 != h4Bucket
        if nextH4
            // M5A: Tres H4 completas consecutivas, sin conectar gaps de mercado.
            bool fullH4 = h4Continuous and h4Count == 48 and h4FirstT == h4Bucket and h4LastEnd == h4Bucket + H4MS
            if fullH4 and not na(h4Prev) and not na(h4Prev2)
                bool chainH4 = h4Prev.valid and h4Prev2.valid and h4Prev2.endAt == h4Prev.startAt and h4Prev.endAt == h4Bucket
                if chainH4
                    if h4Low > h4Prev2.hi
                        f_add_context_fvg(fvgsH4, "H4", 1, h4Low, h4Prev2.hi, h4LastEnd, previousBarEnd == h4LastEnd and previousSessionLast5)
                    else if h4High < h4Prev2.lo
                        f_add_context_fvg(fvgsH4, "H4", -1, h4Prev2.lo, h4High, h4LastEnd, previousBarEnd == h4LastEnd and previousSessionLast5)
            h4Prev2 := h4Prev
            h4Prev := FvgBar.new(h4Bucket, h4LastEnd, h4High, h4Low, fullH4)
            // C3 solo se invalida por el CIERRE de su H4, NO por cierre M5.
            // La mecha o un cierre intrabar M5 que atraviese el extremo de C2
            // no revocan el CRT cuando la C3 H4 finalmente cierra recuperada.
            if c3Active and c3Start == h4Bucket
                bool validC3Close = h4Continuous and h4FirstT == h4Bucket and h4LastEnd == bucketH4 and (isBTC ? (h4Count == 48 and bucketH4 - h4Bucket == H4MS) : h4Count >= minBarsC2)
                bool invalidAtC3Close = validC3Close and not na(c3Invalidation) and (c3Dir == 1 ? h4Close < c3Invalidation : h4Close > c3Invalidation)
                if (invalidAtC3Close or (isBTC and not validC3Close)) and lastC1 == c1Start3
                    lastConfirmedAt := na
                    lastM5 := na
                    lastM15 := na
                    lastModelKind := "NONE"
                    lastPatternValid := false
                    lastModelC1 := na
                c3Active := false
                c3ModelKind := "NONE"
                c3PatternValid := false
                c3ModelC1 := na
                c3C2Start := na

            // M3B.2 INTEGRADO: clasificar solo al cierre H4 de la manipulacion.
            // Una barrida M5 no equivale por si sola a un CRT ratificado.
            if array.size(insideCandidates) > 0
                for ic = 0 to array.size(insideCandidates) - 1
                    InsideCandidate cand = array.get(insideCandidates, ic)
                    bool closesSweepH4 = not na(cand.sweepH4Start) and cand.sweepH4Start == h4Bucket
                    if closesSweepH4 and (cand.state == "SWEPT" or cand.state == "AMBIGUOUS")
                        bool closedH4Complete = h4Continuous and h4FirstT == h4Bucket and h4LastEnd == bucketH4 and h4Count >= minBarsC2
                        if not closedH4Complete
                            cand.state := "DATA_GAP"
                            cand.isPatternValid := false
                        else if cand.state == "AMBIGUOUS" or (cand.sweptLow and cand.sweptHigh)
                            cand.state := "AMBIGUOUS"
                            cand.modelKind := "NONE"
                            cand.isPatternValid := false
                        else
                            bool returnC2 = h4Close >= cand.c2Low and h4Close <= cand.c2High
                            if returnC2
                                cand.state := "RATIFIED"
                                cand.modelKind := cand.sweepDir == 1 ? "INSIDE_RETURN_BULL" : "INSIDE_RETURN_BEAR"
                                cand.isPatternValid := true
                            else if h4Close < cand.c1Low or h4Close > cand.c1High
                                cand.state := "CONTINUATION"
                                cand.modelKind := h4Close < cand.c1Low ? "INSIDE_CONTINUATION_DOWN" : "INSIDE_CONTINUATION_UP"
                                cand.isPatternValid := false // Experimental, sin entradas
                            else
                                // Cierra dentro de C1 pero fuera de C2: sin clasificacion final.
                                cand.state := "UNRESOLVED"
                                cand.modelKind := "NONE"
                                cand.isPatternValid := false

            // ---- M3B.3B: EXPIRACION Y GAPS (DESPUES de clasificar 3B.2) ----
            // No confundir una pausa normal entre sesiones con un hueco M5
            // interno. Contamos cierres H4 verificados, no tiempo UTC / 4h.
            bool insideH4Closed = h4Continuous and h4FirstT == h4Bucket and h4Count >= minBarsC2 and (h4LastEnd == bucketH4 or h4LastEnd == h4Bucket + H4MS)
            if array.size(insideCandidates) > 0
                for ageI = 0 to array.size(insideCandidates) - 1
                    InsideCandidate ageCand = array.get(insideCandidates, ageI)
                    bool awaiting = ageCand.state == "SEEKING" or ageCand.state == "SWEPT"
                    bool eligibleH4 = h4Bucket >= ageCand.c2End
                    if awaiting and eligibleH4
                        if not h4Continuous
                            // Hay velas M5 en esta H4 pero falta continuidad interna.
                            ageCand.state := "DATA_GAP"
                            ageCand.isPatternValid := false
                        else if insideH4Closed
                            if ageCand.state == "SEEKING" and (h4Close < ageCand.c1Low or h4Close > ageCand.c1High)
                                ageCand.state := "INVALIDATED"
                                ageCand.isPatternValid := false
                            else
                                ageCand.h4Life += 1
                                if ageCand.h4Life >= maxH4Life
                                    ageCand.state := "EXPIRED"
                                    ageCand.isPatternValid := false
            // Los estados RATIFIED/CONTINUATION/AMBIGUOUS/UNRESOLVED son
            // terminales: no se reactivan ni consumen mas vida H4.

            // C2 pasa a C3 solo cuando la H4 de C2 CIERRA dentro de C1.
            bool c2Done = c2Active and c2Start == h4Bucket
            if c2Done
                bool inside = h4Close >= c1Low2 and h4Close <= c1High2
                // El modulo 1 permanece OBLIGATORIO para cualquier clasificacion.
                bool validC2Data = h4Continuous and h4FirstT == h4Bucket and h4LastEnd == bucketH4 and (isBTC ? (h4Count == 48 and bucketH4 - h4Bucket == H4MS) : h4Count >= minBarsC2)
                // M3B.1: C2 estrictamente interior a C1 al cierre H4 real.
                // No ratifica Inside, no altera C2-sweep, MSS ni dashboard.
                bool strictH4Inside = validC2Data and h4High < c1High2 and h4Low > c1Low2
                if strictH4Inside
                    string insideId = syminfo.tickerid + "|" + str.tostring(c1Start2) + "|INSIDE"
                    bool alreadyStored = false
                    int candidateCount = array.size(insideCandidates)
                    if candidateCount > 0
                        for j = 0 to candidateCount - 1
                            InsideCandidate existing = array.get(insideCandidates, j)
                            if existing.id == insideId
                                alreadyStored := true
                                break
                    if not alreadyStored
                        InsideCandidate cand = InsideCandidate.new(insideId, c1Start2, c1High2, c1Low2, h4High, h4Low, bucketH4, na, na, 0, false, false, "SEEKING", "NONE", false, 0)
                        // Una memoria llena descarta primero terminales sin
                        // ratificacion, despues SEEKING, SWEPT y finalmente
                        // el RATIFIED mas antiguo si todos son importantes.
                        if array.size(insideCandidates) >= 10
                            int dropIdx = na
                            for tier = 0 to 3
                                if na(dropIdx)
                                    for mi = 0 to array.size(insideCandidates) - 1
                                        string st = array.get(insideCandidates, mi).state
                                        bool discardable = tier == 0 ? (st == "EXPIRED" or st == "INVALIDATED" or st == "DATA_GAP" or st == "AMBIGUOUS" or st == "UNRESOLVED" or st == "CONTINUATION") : tier == 1 ? st == "SEEKING" : tier == 2 ? st == "SWEPT" : st == "RATIFIED"
                                        if discardable
                                            dropIdx := mi
                                            break
                            array.remove(insideCandidates, na(dropIdx) ? 0 : dropIdx)
                        array.push(insideCandidates, cand)
                bool strictSingleSweep = c2Dir != 0 and c2StrictSweep and not c2Double
                bool expansionBull = c2Dir == 1 and h4Close < c1Low2
                bool expansionBear = c2Dir == -1 and h4Close > c1High2
                string classified2C = "NONE"
                if validC2Data and strictSingleSweep
                    if inside and c2Returned
                        classified2C := "CRT_2C_RETURN"
                    else if expansionBull or expansionBear
                        classified2C := "CRT_2C_EXPANSION"  // Solo diagnostico, NUNCA ratificado
                last2CClassification := classified2C
                last2CClassificationAt := bucketH4
                bool validRange = classified2C == "CRT_2C_RETURN"
                bool exactHour3 = hour(bucketH4, TZ) == ((hour(c1Start2, TZ) + 8) % 24)
                if validRange and exactHour3
                    // Registrar el CRT al CIERRE H4 de C2. M5/M15 se
                    // conservan solo si ya se confirmaron en esa C2.
                    lastC1 := c1Start2
                    lastDir := c2Dir
                    lastConfirmedAt := bucketH4
                    lastHi := c1High2
                    lastLo := c1Low2
                    lastSweepAt := c2SweepEnd
                    lastM5 := c2M5At
                    lastM15 := c2M15At
                    lastM5Level := c2Pivot5
                    lastM15Level := c2Pivot15
                    lastM5PivotAt := c2Pivot5At
                    lastM15PivotAt := c2Pivot15At
                    lastModelKind := classified2C
                    lastPatternValid := true
                    lastModelC1 := c1Start2
                    lastC2Start := c2Start
                    c3ModelKind := classified2C
                    c3PatternValid := true
                    c3ModelC1 := c1Start2
                    c3C2Start := c2Start
                    c3Active := true
                    c3Start := bucketH4
                    c1Start3 := c1Start2
                    c1High3 := c1High2
                    c1Low3 := c1Low2
                    c3Dir := c2Dir
                    c3SweepEnd := c2SweepEnd
                    c3Pivot5 := c2Pivot5
                    c3Pivot15 := c2Pivot15
                    c3Pivot5At := c2Pivot5At
                    c3Pivot15At := c2Pivot15At
                    c3M5At := c2M5At
                    c3M15At := c2M15At
                    c3Invalidation := c2Dir == 1 ? h4Low : h4High
                c2Active := false

            // La H4 que acaba de cerrar puede ser C1 del SIGUIENTE ciclo,
            // incluso si la H4 nueva ya es C3 de un ciclo anterior.
            c2Active := false
            c2Dir := 0
            c2StrictSweep := false
            c2Double := false
            c2Returned := false
            c2SweepEnd := na
            c2M5At := na
            c2M15At := na
            c2Pivot5 := na
            c2Pivot15 := na
            c2Pivot5At := na
            c2Pivot15At := na
            int lastC1Hour = hour(h4Bucket, TZ)
            int expectedC2 = (lastC1Hour + 4) % 24
            if f_allowed(lastC1Hour, isIndex) and hour(bucketH4, TZ) == expectedC2 and h4Continuous and h4FirstT == h4Bucket and h4LastEnd == bucketH4 and (isBTC ? (h4Count == 48 and bucketH4 - h4Bucket == H4MS) : h4Count >= minBarsC1) and h4High > h4Low
                c2Active := true
                c2Start := bucketH4
                c1Start2 := h4Bucket
                c1High2 := h4High
                c1Low2 := h4Low

            h4Bucket := bucketH4
            h4High := bH
            h4Low := bL
            h4Close := bC
            h4Count := 1
            h4FirstT := barT
            h4LastEnd := barEnd
            h4Continuous := true
        else if na(h4Bucket)
            h4Bucket := bucketH4
            h4High := bH
            h4Low := bL
            h4Close := bC
            h4Count := 1
            h4FirstT := barT
            h4LastEnd := barEnd
            h4Continuous := true
        else
            h4Continuous := h4Continuous and barT == h4LastEnd
            h4High := math.max(h4High, bH)
            h4Low := math.min(h4Low, bL)
            h4Close := bC
            h4Count += 1
            h4LastEnd := barEnd

    // ---- M4B: ESTADO DE LIQUIDEZ DE LA SESION DE REFERENCIA ----
    // M4A ya procesó esta M5 cerrada. Después del cambio H4,
    // c2Active/c3Active corresponden al H4 actual, incluso en su primera M5.
    bool hasLiquidityRef = prev_valid and not na(prev_timestamp) and not na(prev_PDH) and not na(prev_PDL)
    if not hasLiquidityRef
        liquidityRefAt := na
        pdhTaken := false
        pdlTaken := false
        pdhSweepAt := na
        pdlSweepAt := na
        pdhSweepStage := 0
        pdlSweepStage := 0
        pdhSweepC1 := na
        pdlSweepC1 := na
    else
        if na(liquidityRefAt) or liquidityRefAt != prev_timestamp
            liquidityRefAt := prev_timestamp
            pdhTaken := false
            pdlTaken := false
            pdhSweepAt := na
            pdlSweepAt := na
            pdhSweepStage := 0
            pdlSweepStage := 0
            pdhSweepC1 := na
            pdlSweepC1 := na
        // No comparar velas del dia anterior con el nuevo PDH/PDL,
        // ni after-hours con niveles RTH. Solo M5 cerrada elegible.
        bool eligibleLiquidityBar = pdhInSession and not na(barT) and not na(barEnd) and barT >= liquidityRefAt
        if eligibleLiquidityBar
            int eventStage = c3Active and bucketH4 == c3Start ? 3 : c2Active and bucketH4 == c2Start ? 2 : 0
            int eventC1 = eventStage == 3 ? c1Start3 : eventStage == 2 ? c1Start2 : na
            if not pdhTaken and bH > prev_PDH
                pdhTaken := true
                pdhSweepAt := barEnd
                pdhSweepStage := eventStage
                pdhSweepC1 := eventC1
            if not pdlTaken and bL < prev_PDL
                pdlTaken := true
                pdlSweepAt := barEnd
                pdlSweepStage := eventStage
                pdlSweepC1 := eventC1

    // ---- M3B.2 INTEGRADO: PRIMERA BARRIDA M5 DE INSIDE BAR ----
    // M5 cerrada [1]; solo despues del cierre H4 de C2.
    // Se rastrean ambos extremos durante una MISMA H4 de manipulacion.
    if not na(barT) and not na(bucketH4) and array.size(insideCandidates) > 0
        for ic = 0 to array.size(insideCandidates) - 1
            InsideCandidate cand = array.get(insideCandidates, ic)
            if cand.state == "SEEKING" and barT >= cand.c2End
                bool brokeLow = bL < cand.c2Low
                bool brokeHigh = bH > cand.c2High
                if brokeLow or brokeHigh
                    cand.sweepAt := barEnd
                    cand.sweepH4Start := bucketH4
                    cand.sweptLow := brokeLow
                    cand.sweptHigh := brokeHigh
                    cand.sweepDir := brokeLow and not brokeHigh ? 1 : brokeHigh and not brokeLow ? -1 : 0
                    cand.state := brokeLow and brokeHigh ? "AMBIGUOUS" : "SWEPT"
            else if cand.state == "SWEPT" and bucketH4 == cand.sweepH4Start
                cand.sweptLow := cand.sweptLow or bL < cand.c2Low
                cand.sweptHigh := cand.sweptHigh or bH > cand.c2High
                if cand.sweptLow and cand.sweptHigh
                    cand.sweepDir := 0
                    cand.state := "AMBIGUOUS"

    // ---- M5B CANDIDATO: SEGUIMIENTO FVG H1/H4 EN ESTA M5 CERRADA ----
    // Va despues de crear FVG nuevos en el cambio de H1/H4: permite que
    // la primera M5 de la siguiente ventana toque el FVG recien confirmado.
    if not na(barT) and not na(barEnd)
        if array.size(fvgsH1) > 0
            for fi = 0 to array.size(fvgsH1) - 1
                FvgContext zoneH1 = array.get(fvgsH1, fi)
                f_update_context_fvg(zoneH1, barT, barEnd, bL, bH, bC, firstSession5, lastSession5)
        if array.size(fvgsH4) > 0
            for fi = 0 to array.size(fvgsH4) - 1
                FvgContext zoneH4 = array.get(fvgsH4, fi)
                f_update_context_fvg(zoneH4, barT, barEnd, bL, bH, bC, firstSession5, lastSession5)

    // ---- BARRIDA Y RETORNO C2 (velas M5 cerradas) ----
    // QA: tocar CRH/CRL (igualdad) solo puede crear un C2 PROVISIONAL.
    // Direccion y c2Double definitivos dependen EXCLUSIVAMENTE de barridas estrictas.
    if c2Active and not na(barT) and bucketH4 == c2Start
        bool strictLow = bL < c1Low2
        bool strictHigh = bH > c1High2
        bool touchLow = bL <= c1Low2
        bool touchHigh = bH >= c1High2
        // Dos extremos estrictamente barridos, incluso en una unica M5.
        if strictLow and strictHigh
            c2Double := true
        if c2Dir == 0 and not c2Double
            if strictLow != strictHigh
                c2Dir := strictLow ? 1 : -1
            else if allowTouches and touchLow != touchHigh
                // Toque exacto unilateral: provisional, sin MSS confirmado.
                c2Dir := touchLow ? 1 : -1
        if c2Dir != 0 and not c2Double
            bool chosenStrict = c2Dir == 1 ? strictLow : strictHigh
            bool oppositeStrict = c2Dir == 1 ? strictHigh : strictLow
            // Si solo hubo toque preliminar, la PRIMERA barrida REAL elige
            // la direccion, aun si ocurre en el lado opuesto al toque.
            if not c2StrictSweep and oppositeStrict and not chosenStrict
                c2Dir := -c2Dir
                chosenStrict := true
                oppositeStrict := false
                c2Returned := false
            // Descartar UNICAMENTE cuando se han barrido estrictamente
            // ambos limites; un toque exacto del otro lado NO basta.
            if c2StrictSweep and oppositeStrict
                c2Double := true
            if not c2Double
                bool firstStrictSweep = not c2StrictSweep and chosenStrict
                if firstStrictSweep
                    c2StrictSweep := true
                    c2SweepEnd := barEnd
                    c2Returned := false
                    bool lowSide = c2Dir == 1
                    bool fresh5 = lowSide ? (not na(lastHigh5At) and lastHigh5At < barT and barT - lastHigh5At <= maxPivotAge * 60000) : (not na(lastLow5At) and lastLow5At < barT and barT - lastLow5At <= maxPivotAge * 60000)
                    bool fresh15 = lowSide ? (not na(lastHigh15At) and lastHigh15At < barT and barT - lastHigh15At <= maxPivotAge * 60000) : (not na(lastLow15At) and lastLow15At < barT and barT - lastLow15At <= maxPivotAge * 60000)
                    c2Pivot5 := fresh5 ? (lowSide ? lastHigh5 : lastLow5) : na
                    c2Pivot15 := fresh15 ? (lowSide ? lastHigh15 : lastLow15) : na
                    c2Pivot5At := fresh5 ? (lowSide ? lastHigh5At : lastLow5At) : na
                    c2Pivot15At := fresh15 ? (lowSide ? lastHigh15At : lastLow15At) : na
                c2Returned := c2Returned or (bC >= c1Low2 and bC <= c1High2)
                // MSS M5 no puede ocurrir ANTES NI DURANTE la vela de barrida.
                bool cross5 = c2Dir == 1 ? (not na(c2Pivot5) and bC > c2Pivot5 and prevClose <= c2Pivot5) : (not na(c2Pivot5) and bC < c2Pivot5 and prevClose >= c2Pivot5)
                if c2StrictSweep and previousM5Continuous and barT >= c2SweepEnd and cross5 and na(c2M5At)
                    c2M5At := barEnd

    // ---- MSS en C3 ratificada (misma direccion, mismo swing congelado) ----
    if c3Active and not na(barT) and bucketH4 == c3Start
        // No invalidar aqui por una vela M5. Solo el cierre H4 completo
        // de C3 puede atravesar la mecha de C2 para invalidar el CRT.
        bool cross5c3 = c3Dir == 1 ? (not na(c3Pivot5) and bC > c3Pivot5 and prevClose <= c3Pivot5) : (not na(c3Pivot5) and bC < c3Pivot5 and prevClose >= c3Pivot5)
        if c3Active and previousM5Continuous and barEnd > c3Start and barEnd > c3SweepEnd and cross5c3 and na(c3M5At)
            c3M5At := barEnd
            if lastC1 == c1Start3
                lastM5 := barEnd

    // Actualizar swing M5 DESPUES de la barrida: no congelar un pivot que
    // acababa de confirmarse simultaneamente con la vela de manipulacion.
    if not na(pH5)
        lastHigh5 := pH5
        lastHigh5At := time[right5 + 1]
    if not na(pL5)
        lastLow5 := pL5
        lastLow5At := time[right5 + 1]

    // Priorizar el CRT ratificado en C3 sobre una NUEVA C2 provisional
    // superpuesta. La nueva C2 permanece registrada internamente y se
    // mostrara cuando termine la C3, si aun continua vigente.
    bool partial = showCandidate and c2Active and c2Dir != 0 and not c2Double and (showAtSweep or c2Returned)
    bool recent = keepLastCRT and not na(lastConfirmedAt) and barEnd >= lastConfirmedAt and barEnd - lastConfirmedAt <= keepHours * 3600000
    int outPhase = c3Active ? 2 : partial ? 1 : recent ? 3 : 0
    int outDir = outPhase == 2 ? c3Dir : outPhase == 1 ? c2Dir : outPhase == 3 ? lastDir : 0
    int outC1 = outPhase == 2 ? c1Start3 : outPhase == 1 ? c1Start2 : outPhase == 3 ? lastC1 : na
    int outC2Start = outPhase == 2 ? c3C2Start : outPhase == 1 ? c2Start : outPhase == 3 ? lastC2Start : na
    float outHi = outPhase == 2 ? c1High3 : outPhase == 1 ? c1High2 : outPhase == 3 ? lastHi : na
    float outLo = outPhase == 2 ? c1Low3 : outPhase == 1 ? c1Low2 : outPhase == 3 ? lastLo : na
    int outSweep = outPhase == 2 ? c3SweepEnd : outPhase == 1 ? c2SweepEnd : outPhase == 3 ? lastSweepAt : na
    int outM5At = outPhase == 2 ? c3M5At : outPhase == 1 ? c2M5At : outPhase == 3 ? lastM5 : na
    int outM15At = outPhase == 2 ? c3M15At : outPhase == 1 ? c2M15At : outPhase == 3 ? lastM15 : na
    float outM5Level = outPhase == 2 ? c3Pivot5 : outPhase == 1 ? c2Pivot5 : outPhase == 3 ? lastM5Level : na
    float outM15Level = outPhase == 2 ? c3Pivot15 : outPhase == 1 ? c2Pivot15 : outPhase == 3 ? lastM15Level : na
    int outM5PivotAt = outPhase == 2 ? c3Pivot5At : outPhase == 1 ? c2Pivot5At : outPhase == 3 ? lastM5PivotAt : na
    int outM15PivotAt = outPhase == 2 ? c3Pivot15At : outPhase == 1 ? c2Pivot15At : outPhase == 3 ? lastM15PivotAt : na
    // El modelo es metadato independiente; phase y ticks MSS no cambian.
    bool c3IdentityOK = outPhase == 2 and c3ModelC1 == outC1
    bool lastIdentityOK = outPhase == 3 and lastModelC1 == outC1
    bool rejectedRecent = outPhase == 0 and last2CClassification == "CRT_2C_EXPANSION" and not na(last2CClassificationAt) and barEnd >= last2CClassificationAt and barEnd - last2CClassificationAt < H4MS
    string outModel = c3IdentityOK ? c3ModelKind : outPhase == 1 ? "PENDING" : lastIdentityOK ? lastModelKind : rejectedRecent ? "CRT_2C_EXPANSION" : "NONE"
    bool outValid = c3IdentityOK ? c3PatternValid : lastIdentityOK ? lastPatternValid : false
    string outPdhStatus = hasLiquidityRef ? (pdhTaken ? "TAKEN" : "INTACTO") : "N/D"
    string outPdlStatus = hasLiquidityRef ? (pdlTaken ? "TAKEN" : "INTACTO") : "N/D"
    bool pdhInDisplayedCRT = pdhTaken and outPhase > 0 and pdhSweepC1 == outC1 and (pdhSweepStage == 2 or pdhSweepStage == 3)
    bool pdlInDisplayedCRT = pdlTaken and outPhase > 0 and pdlSweepC1 == outC1 and (pdlSweepStage == 2 or pdlSweepStage == 3)
    // QA SOLO: FVG mas recientemente FORMADO en cada TF (no el ultimo vigente).
    int qaCountH1 = array.size(fvgsH1)
    int qaCodeH1 = -1
    float qaTopH1 = na
    float qaBottomH1 = na
    float qaPenH1 = na
    int qaFormH1 = na
    int qaGapH1 = na
    if qaCountH1 > 0
        FvgContext qaH1 = array.get(fvgsH1, qaCountH1 - 1)
        qaCodeH1 := f_fvg_code(qaH1.state)
        qaTopH1 := qaH1.top
        qaBottomH1 := qaH1.bottom
        qaPenH1 := qaH1.maxPenetration * 100.0
        qaFormH1 := qaH1.formedAt
        qaGapH1 := qaH1.gapObservedAt
    int qaCountH4 = array.size(fvgsH4)
    int qaCodeH4 = -1
    float qaTopH4 = na
    float qaBottomH4 = na
    float qaPenH4 = na
    int qaFormH4 = na
    int qaGapH4 = na
    if qaCountH4 > 0
        FvgContext qaH4 = array.get(fvgsH4, qaCountH4 - 1)
        qaCodeH4 := f_fvg_code(qaH4.state)
        qaTopH4 := qaH4.top
        qaBottomH4 := qaH4.bottom
        qaPenH4 := qaH4.maxPenetration * 100.0
        qaFormH4 := qaH4.formedAt
        qaGapH4 := qaH4.gapObservedAt
    Signal.new(outPhase, outDir, outC1, outHi, outLo, outSweep, outM5At, outM15At, outM5Level, outM15Level, outM5PivotAt, outM15PivotAt, outModel, outValid, outPdhStatus, outPdlStatus, pdhSweepAt, pdlSweepAt, pdhSweepStage, pdlSweepStage, pdhSweepC1, pdlSweepC1, outC2Start, outPhase == 2 ? c3Start : na, liquidityRefAt, pdhInDisplayedCRT, pdlInDisplayedCRT, qaCountH1, qaCodeH1, qaTopH1, qaBottomH1, qaPenH1, qaFormH1, qaGapH1, qaCountH4, qaCodeH4, qaTopH4, qaBottomH4, qaPenH4, qaFormH4, qaGapH4, bucketH4, h4Count)

// ciclo C2/C3 mostrado. Evita atribuir al CRT una barrida ajena.
f_liqText(string status, int sweepAt, int stage, int sweepC1, int shownC1, int shownC2) =>
    string result = status == "N/D" ? "N/D" : status == "INTACTO" ? "INT" : "T OTR"
    if status == "TAKEN"
        if not na(shownC1) and sweepC1 == shownC1 and stage == 2
            result := "T C2"
        else if not na(shownC1) and sweepC1 == shownC1 and stage == 3
            result := "T C3"
        else if not na(shownC2) and not na(sweepAt) and sweepAt <= shownC2
            result := "T PRE"
    result
