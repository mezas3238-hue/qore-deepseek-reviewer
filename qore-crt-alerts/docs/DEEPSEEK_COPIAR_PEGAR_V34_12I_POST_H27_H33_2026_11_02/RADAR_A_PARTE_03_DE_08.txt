                        qaRetestSourceMssAt := qaMssAt

                prev15Close := na
                prev15End := na
                array.clear(highs15)
                array.clear(lows15)
                array.clear(times15)

            m15Low := bL
            m15Close := bC

            m15Start := bucket15

                array.clear(lows15)
                array.clear(times15)

                lastLow15 := na

                lastLow15At := na
            else if m15Count >= 0

    if not na(barT) and not na(bucketH1)
        bool nextH1 = not na(h1Bucket) and bucketH1 != h1Bucket
        if nextH1
            bool fullH1 = h1Continuous and h1Count == 12 and h1FirstT == h1Bucket and h1LastEnd == h1Bucket + 3600000
            if fullH1 and not na(h1Prev) and not na(h1Prev2)
                bool chainH1 = h1Prev.valid and h1Prev2.valid and h1Prev2.endAt == h1Prev.startAt and h1Prev.endAt == h1Bucket
                if chainH1
                    if h1Low > h1Prev2.hi
                        f_add_context_fvg(fvgsH1, "H1", 1, h1Low, h1Prev2.hi, h1LastEnd, previousBarEnd == h1LastEnd and previousSessionLast5)
                    else if h1High < h1Prev2.lo
                        f_add_context_fvg(fvgsH1, "H1", -1, h1Prev2.lo, h1High, h1LastEnd, previousBarEnd == h1LastEnd and previousSessionLast5)
            h1Prev2 := h1Prev
            h1Prev := FvgBar.new(h1Bucket, h1LastEnd, h1High, h1Low, fullH1)
            h1Bucket := bucketH1
            h1FirstT := barT
            h1LastEnd := barEnd
            h1High := bH
            h1Low := bL
            h1Count := 1
            h1Continuous := true
        else if na(h1Bucket)
            h1Bucket := bucketH1
            h1FirstT := barT
            h1LastEnd := barEnd
            h1High := bH
            h1Low := bL
            h1Count := 1
            h1Continuous := true
        else

    if not na(barT) and not na(bucketH4)
        bool nextH4 = not na(h4Bucket) and bucketH4 != h4Bucket
        if nextH4
            // QA: registrar cuántos bloques H4 del mismo feed se han
            // observado; no confundir respuesta de request.security con
            // presencia de C1 válida ni inventar formaciones CRT.
            qaAuditC1H4Closed += 1

            // Time deltas are based on the same M5 feed, not chart M15/H4.
            qaH4LastBars := h4Count

            if h4LastEnd != h4Bucket + H4MS
                qaH4BadEnd += 1

                    // Hueco dentro de un bloque que sí tenía M5.
                    qaAuditH4Internal += 1
                else
                    // Inicio/fin incompleto (historial o horario de
                    // mercado); no atribuir a pérdida del proveedor.
                    qaAuditH4Session += 1
            if fullH4 and not na(h4Prev) and not na(h4Prev2)

                    if h4Low > h4Prev2.hi

            // QA MULTI4: barrida y retorno se distribuyen en DOS H4

                            bool validTimeC1 = f_allowed(hour(beginH4.startAt, TZ), isIndex)
                            bool sweepLowN = endH4.lo < beginH4.lo and endH4.hi <= beginH4.hi
                            bool sweepHighN = endH4.hi > beginH4.hi and endH4.lo >= beginH4.lo

                            if consecutive and internalBars and validTimeC1

                                multiNSawDouble := multiNSawDouble or sweptBoth
                                multiNSawOutside := multiNSawOutside or (sweepLowN != sweepHighN) and not returnN
