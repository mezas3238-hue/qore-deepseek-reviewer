f_monitorOwnMss(int slot) =>
    bool coreOrQa = array.get(validCoreFlags,slot) or array.get(qaCoreObservedFlags,slot)
    int c1T = array.get(qaC1BySlot,slot)
    int sweepT = array.get(qaSweepBySlot,slot)
    int ratT = array.get(qaRatBySlot,slot)
    int p5 = array.get(m5PivotTimes,slot)
    int b5 = array.get(m5Times,slot)
    int p15 = array.get(qaPivot15AtBySlot,slot)
    int b15 = array.get(m15Times,slot)
    int lastClosed15 = array.get(qaClosed15BySlot,slot)
    int phase = array.get(qaSourcePhaseBySlot,slot)
    bool cycleRoot = coreOrQa and not na(c1T) and not na(sweepT) and not na(ratT) and sweepT > c1T and ratT >= sweepT
    bool phaseOK = not na(phase) and phase >= 0 and phase <= 4
    bool m5Confirmed = cycleRoot and phaseOK and not na(p5) and p5 < sweepT and b5 > sweepT and b5 % MIN5 == phase * 60000
    bool m15Confirmed = cycleRoot and phaseOK and not na(p15) and p15 < sweepT and b15 > sweepT and not na(lastClosed15) and b15 <= lastClosed15 and f_m15_phase_ok(b15,phase)
    [m5Confirmed,m15Confirmed]
