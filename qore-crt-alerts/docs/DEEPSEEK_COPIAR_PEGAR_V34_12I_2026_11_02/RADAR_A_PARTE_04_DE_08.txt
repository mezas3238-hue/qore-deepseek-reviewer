                    bool multiNAuditable = false
                    bool multiNMatched = false
                    bool multiNSawDouble = false
                    bool multiNSawOutside = false
                    bool multiNSawNoTerminal = false
                    // Un mismo C1 puede producir varias interpretaciones.
                    // Preferir la C1 mas reciente (N menor) y registrar
                    // solo una observacion QA por vela terminal.
                    for consolidationCount = 2 to 5
                        int needed = consolidationCount + 2
                        if needed <= nHistory
                            int c1Idx = nHistory - needed
                            MultiH4 beginH4 = array.get(qaMultiNHistory, c1Idx)
                            MultiH4 endH4 = array.get(qaMultiNHistory, nHistory - 1)
                            bool consecutive = true
                            bool internalBars = true
                            MultiH4 priorH4 = beginH4
                            for j = c1Idx + 1 to nHistory - 1
                                MultiH4 nextBar = array.get(qaMultiNHistory, j)
                                consecutive := consecutive and priorH4.endAt == nextBar.startAt
                                priorH4 := nextBar
                            for j = c1Idx + 1 to nHistory - 2
                                MultiH4 consolidate = array.get(qaMultiNHistory, j)
                                internalBars := internalBars and consolidate.hi < beginH4.hi and consolidate.lo > beginH4.lo
                            bool validTimeC1 = f_allowed(hour(beginH4.startAt, TZ), isIndex)
                            bool sweepLowN = endH4.lo < beginH4.lo and endH4.hi <= beginH4.hi
                            bool sweepHighN = endH4.hi > beginH4.hi and endH4.lo >= beginH4.lo
                            bool returnN = endH4.cl >= beginH4.lo and endH4.cl <= beginH4.hi
                            if consecutive and internalBars and validTimeC1
                                multiNAuditable := true
                                bool sweptBoth = endH4.lo < beginH4.lo and endH4.hi > beginH4.hi
                                multiNSawDouble := multiNSawDouble or sweptBoth
                                multiNSawOutside := multiNSawOutside or (sweepLowN != sweepHighN) and not returnN
                                multiNSawNoTerminal := multiNSawNoTerminal or (not sweepLowN and not sweepHighN and not sweptBoth)
                            if consecutive and internalBars and validTimeC1 and (sweepLowN != sweepHighN) and returnN
                                multiNMatched := true
                                qaAuditMN += 1
                                qaLastMultiNAt := endH4.endAt
                                qaMultiNKind := sweepLowN ? "MULTI-N BULL QA" : "MULTI-N BEAR QA"
                                qaMultiNAt := endH4.endAt
                                qaMultiNC1 := beginH4.startAt
                                qaMultiNDir := sweepLowN ? 1 : -1
                                // Solo extraer MSS del MISMO candidato C1
                                // y H4 terminal, con barrida no ambigua.
                                qaMultiNSweepAt := na
                                qaMultiNPivot15 := na
                                qaMultiNPivot15At := na
                                qaMultiNMss15At := na
                                qaMultiNMss5At := na
                                qaMultiNPivot5 := na
                                qaMultiNPivot5At := na
                                if array.size(qaMultiNMssCandidates) > 0
                                    for k = 0 to array.size(qaMultiNMssCandidates) - 1
                                        MultiNMssCandidate qaOwn = array.get(qaMultiNMssCandidates, k)
                                        if qaOwn.c1Start == qaMultiNC1 and qaOwn.terminalH4Start == h4Bucket and qaOwn.direction == qaMultiNDir and not qaOwn.doubleSweep
                                            qaMultiNSweepAt := qaOwn.sweepAt
                                            qaMultiNPivot15 := qaOwn.pivot15
                                            qaMultiNPivot15At := qaOwn.pivot15At
                                            qaMultiNMss15At := qaOwn.mss15At
                                            qaMultiNPivot5 := qaOwn.pivot5
                                            qaMultiNPivot5At := qaOwn.pivot5At
                                            qaMultiNMss5At := qaOwn.mss5At
                                            break
                                break
                    // Solo UNA razon por vela H4 terminal: el mismo cierre
                    // puede cumplir varias N candidatas. No contar intentos
                    // intrabar M5 ni la misma barra dos veces.
                    if multiNAuditable and not multiNMatched
                        if multiNSawDouble
                            qaAuditMNDouble += 1
                        else if multiNSawOutside
                            qaAuditMNOutside += 1
                        else if multiNSawNoTerminal
                            qaAuditMNNoTerminal += 1
            // Los candidatos pertenecen exclusivamente a la H4 que
            // ACABA de cerrar; no se arrastran a un ciclo nuevo.
            array.clear(qaMultiNMssCandidates)
            // C3 solo se invalida por el CIERRE de su H4, NO por cierre M5.
            // La mecha o un cierre intrabar M5 que atraviese el extremo de C2
            // no revocan el CRT cuando la C3 H4 finalmente cierra recuperada.
            if c3Active and c3Start == h4Bucket
                bool validC3Close = fullH4
                bool invalidAtC3Close = validC3Close and not na(c3Invalidation) and (c3Dir == 1 ? h4Close < c3Invalidation : h4Close > c3Invalidation)
                if invalidAtC3Close or not validC3Close
                    qaAuditC3Invalid += 1
                else if validC3Close
                    qaAudit3C += 1
                    qaLastConfirmed3CAt := h4LastEnd
                if (invalidAtC3Close or not validC3Close) and lastC1 == c1Start3
                    lastConfirmedAt := na
                    lastM5 := na
                    lastM15 := na
                    lastModelKind := "NONE"
                    lastPatternValid := false
                    lastModelC1 := na
                c3Active := false
                c3ModelKind := "NONE"
                c3PatternValid := false
                c3ModelC1 := na
                c3C2Start := na

            // M3B.2 INTEGRADO: clasificar solo al cierre H4 de la manipulacion.
            // Una barrida M5 no equivale por si sola a un CRT ratificado.
            if array.size(insideCandidates) > 0
                for ic = 0 to array.size(insideCandidates) - 1
                    InsideCandidate cand = array.get(insideCandidates, ic)
                    bool closesSweepH4 = not na(cand.sweepH4Start) and cand.sweepH4Start == h4Bucket
                    if closesSweepH4 and (cand.state == "SWEPT" or cand.state == "AMBIGUOUS")
                        bool closedH4Complete = fullH4
                        if not closedH4Complete
                            qaAuditInsideData += 1
                            cand.state := "DATA_GAP"
                            cand.isPatternValid := false
                        else if cand.state == "AMBIGUOUS" or (cand.sweptLow and cand.sweptHigh)
                            qaAuditInsideDouble += 1
                            cand.state := "AMBIGUOUS"
                            cand.modelKind := "NONE"
                            cand.isPatternValid := false
                        else
                            bool returnC2 = h4Close >= cand.c2Low and h4Close <= cand.c2High
                            if returnC2
                                qaAuditInside += 1
                                qaLastInsideReturnAt := h4LastEnd
                                cand.state := "RATIFIED"
                                cand.modelKind := cand.sweepDir == 1 ? "INSIDE_RETURN_BULL" : "INSIDE_RETURN_BEAR"
                                cand.isPatternValid := true
                            else if h4Close < cand.c1Low or h4Close > cand.c1High
                                qaAuditInsideOther += 1
                                qaLastInsideContinuationAt := h4LastEnd
                                cand.state := "CONTINUATION"
                                cand.modelKind := h4Close < cand.c1Low ? "INSIDE_CONTINUATION_DOWN" : "INSIDE_CONTINUATION_UP"
                                cand.isPatternValid := false // Experimental, sin entradas
                            else
                                // Cierra dentro de C1 pero fuera de C2.
                                qaAuditInsideOther += 1
                                cand.state := "UNRESOLVED"
                                cand.modelKind := "NONE"
                                cand.isPatternValid := false

            // ---- M3B.3B: EXPIRACION Y GAPS (DESPUES de clasificar 3B.2) ----
            // No confundir una pausa normal entre sesiones con un hueco M5
            // interno. Contamos cierres H4 verificados, no tiempo UTC / 4h.
            bool insideH4Closed = fullH4
            if array.size(insideCandidates) > 0
                for ageI = 0 to array.size(insideCandidates) - 1
                    InsideCandidate ageCand = array.get(insideCandidates, ageI)
                    bool awaiting = ageCand.state == "SEEKING" or ageCand.state == "SWEPT"
                    bool eligibleH4 = h4Bucket >= ageCand.c2End
                    if awaiting and eligibleH4
                        if not h4Continuous
                            // Hay velas M5 en esta H4 pero falta continuidad interna.
                            qaAuditInsideData += 1
                            ageCand.state := "DATA_GAP"
                            ageCand.isPatternValid := false
                        else if insideH4Closed
                            if ageCand.state == "SEEKING" and (h4Close < ageCand.c1Low or h4Close > ageCand.c1High)
                                qaAuditInsideOther += 1
                                ageCand.state := "INVALIDATED"
                                ageCand.isPatternValid := false
                            else
                                ageCand.h4Life += 1
                                if ageCand.h4Life >= maxH4Life
                                    qaAuditInsideExpire += 1
                                    ageCand.state := "EXPIRED"
                                    ageCand.isPatternValid := false
            // Los estados RATIFIED/CONTINUATION/AMBIGUOUS/UNRESOLVED son
            // terminales: no se reactivan ni consumen mas vida H4.

            // C2 pasa a C3 solo cuando la H4 de C2 CIERRA dentro de C1.
            bool c2Done = c2Active and c2Start == h4Bucket
            if c2Done
                bool inside = h4Close >= c1Low2 and h4Close <= c1High2
                // El modulo 1 permanece OBLIGATORIO para cualquier clasificacion.
                bool validC2Data = fullH4
                // M3B.1: C2 estrictamente interior a C1 al cierre H4 real.
                // No ratifica Inside, no altera C2-sweep, MSS ni dashboard.
                bool strictH4Inside = validC2Data and h4High < c1High2 and h4Low > c1Low2
                if strictH4Inside
                    string insideId = syminfo.tickerid + "|" + str.tostring(c1Start2) + "|INSIDE"
                    bool alreadyStored = false
                    int candidateCount = array.size(insideCandidates)
                    if candidateCount > 0
                        for j = 0 to candidateCount - 1
                            InsideCandidate existing = array.get(insideCandidates, j)
                            if existing.id == insideId
                                alreadyStored := true
                                break
                    if not alreadyStored
                        InsideCandidate cand = InsideCandidate.new(insideId, c1Start2, c1High2, c1Low2, h4High, h4Low, bucketH4, na, na, 0, false, false, "SEEKING", "NONE", false, 0, na, na, na, na, na, na)
                        // Una memoria llena descarta primero terminales sin
                        // ratificacion, despues SEEKING, SWEPT y finalmente
                        // el RATIFIED mas antiguo si todos son importantes.
                        if array.size(insideCandidates) >= 10
                            int dropIdx = na
                            for tier = 0 to 3
                                if na(dropIdx)
                                    for mi = 0 to array.size(insideCandidates) - 1
                                        string st = array.get(insideCandidates, mi).state
                                        bool discardable = tier == 0 ? (st == "EXPIRED" or st == "INVALIDATED" or st == "DATA_GAP" or st == "AMBIGUOUS" or st == "UNRESOLVED" or st == "CONTINUATION") : tier == 1 ? st == "SEEKING" : tier == 2 ? st == "SWEPT" : st == "RATIFIED"
                                        if discardable
                                            dropIdx := mi
                                            break
                            array.remove(insideCandidates, na(dropIdx) ? 0 : dropIdx)
                        array.push(insideCandidates, cand)
                bool strictSingleSweep = c2Dir != 0 and c2StrictSweep and not c2Double
                bool expansionBull = c2Dir == 1 and h4Close < c1Low2
                bool expansionBear = c2Dir == -1 and h4Close > c1High2
                string classified2C = "NONE"
                if validC2Data and strictSingleSweep
                    if inside and c2Returned
                        classified2C := "CRT_2C_RETURN"
                    else if expansionBull or expansionBear
                        classified2C := "CRT_2C_EXPANSION"  // Solo diagnostico, NUNCA ratificado
                last2CClassification := classified2C
                last2CClassificationAt := bucketH4
                if classified2C == "CRT_2C_EXPANSION"
                    qaLastRejectedExpansionAt := h4LastEnd
                bool validRange = classified2C == "CRT_2C_RETURN"
                bool exactHour3 = hour(bucketH4, TZ) == ((hour(c1Start2, TZ) + 8) % 24)
                // Una causa por cierre de C2; son descartes H4, no
                // señales. Incluye BTC, FX e indices del feed actual.
                if not validC2Data
                    qaAuditC2Data += 1
                else if c2Double
                    qaAuditC2Double += 1
                else if not strictSingleSweep
                    qaAuditC2NoSweep += 1
                else if not inside or not c2Returned
                    qaAuditC2Outside += 1
                else if not exactHour3
                    qaAuditC2Hour += 1
                if isBTC
                    qaLastC2Gate := not validC2Data ? "C2 M5 NO 48" : c2Double ? "C2 DOBLE BARRIDA" : not strictSingleSweep ? "C2 SIN BARRIDA" : not inside ? "C2 CIERRE FUERA" : not c2Returned ? "C2 SIN RETORNO" : not exactHour3 ? "C3 HORA INVALIDA" : "CRT C2 RATIFICADO"
                if validRange and exactHour3
                    qaAudit2C += 1
                    qaLastRatified2CAt := h4LastEnd
                    if isBTC
                        qaC2Ratified += 1
                    // Registrar el CRT al CIERRE H4 de C2. M5/M15 se
                    // conservan solo si ya se confirmaron en esa C2.
                    lastC1 := c1Start2
                    lastDir := c2Dir
                    lastConfirmedAt := bucketH4
                    lastHi := c1High2
                    lastLo := c1Low2
                    lastSweepAt := c2SweepEnd
                    lastM5 := c2M5At
                    lastM15 := c2M15At
                    lastM5Level := c2Pivot5
                    lastM15Level := c2Pivot15
                    lastM5PivotAt := c2Pivot5At
                    lastM15PivotAt := c2Pivot15At
                    lastModelKind := classified2C
                    lastPatternValid := true
                    lastModelC1 := c1Start2
                    lastC2Start := c2Start
                    c3ModelKind := classified2C
                    c3PatternValid := true
                    c3ModelC1 := c1Start2
                    c3C2Start := c2Start
                    c3Active := true
                    c3Start := bucketH4
                    c1Start3 := c1Start2
                    c1High3 := c1High2
                    c1Low3 := c1Low2
                    c3Dir := c2Dir
                    c3SweepEnd := c2SweepEnd
                    c3Pivot5 := c2Pivot5
                    c3Pivot15 := c2Pivot15
                    c3Pivot5At := c2Pivot5At
                    c3Pivot15At := c2Pivot15At
                    c3M5At := c2M5At
                    c3M15At := c2M15At
                    c3Invalidation := c2Dir == 1 ? h4Low : h4High
                c2Active := false

            // La H4 que acaba de cerrar puede ser C1 del SIGUIENTE ciclo,
            // incluso si la H4 nueva ya es C3 de un ciclo anterior.
            c2Active := false
            c2Dir := 0
            c2StrictSweep := false
            c2Double := false
            c2Returned := false
            c2SweepEnd := na
            c2M5At := na
            c2M15At := na
            c2Pivot5 := na
            c2Pivot15 := na
            c2Pivot5At := na
            c2Pivot15At := na
            int lastC1Hour = hour(h4Bucket, TZ)
            int expectedC2 = (lastC1Hour + 4) % 24
            bool c1AllowedHour = f_allowed(lastC1Hour, isIndex)
            bool c1AlignedNext = hour(bucketH4, TZ) == expectedC2
            bool c1CompleteData = fullH4
            bool c1HasRange = h4High > h4Low
            // Cada cierre elegible por HORA aporta exactamente una
            // clasificación: mala ancla, datos, rango o C1 apta.
            if c1AllowedHour
                qaAuditC1HourCandidates += 1
                if not c1AlignedNext
                    qaAuditC1BadClock += 1
                else if not c1CompleteData
                    qaAuditC1BadData += 1
                else if not c1HasRange
                    qaAuditC1BadRange += 1
            if c1AllowedHour and c1AlignedNext and c1CompleteData and c1HasRange
                c2Active := true
                // BUG corregido V34.10v: este contador ES por feed.
                // Antes solo incrementaba en BTC pero audit2C sí
                // sumaba Forex/índices, generando el imposible 5/0.
                qaC1Ready += 1
                if isBTC
                    qaLastC2Gate := "C1 VALIDA ESPERA C2"
                c2Start := bucketH4
                c1Start2 := h4Bucket
                c1High2 := h4High
                c1Low2 := h4Low

            h4Bucket := bucketH4
            h4High := bH
            h4Low := bL
            h4Close := bC
            h4Count := 1
            h4FirstT := barT
            h4LastEnd := barEnd
            h4Continuous := true
        else if na(h4Bucket)
            h4Bucket := bucketH4
            h4High := bH
            h4Low := bL
            h4Close := bC
            h4Count := 1
            h4FirstT := barT
            h4LastEnd := barEnd
            h4Continuous := true
        else
            h4Continuous := h4Continuous and barT == h4LastEnd
            h4High := math.max(h4High, bH)
            h4Low := math.min(h4Low, bL)
            h4Close := bC
            h4Count += 1
            h4LastEnd := barEnd

    // ---- M4B: ESTADO DE LIQUIDEZ DE LA SESION DE REFERENCIA ----
    // M4A ya procesó esta M5 cerrada. Después del cambio H4,
    // c2Active/c3Active corresponden al H4 actual, incluso en su primera M5.
    bool hasLiquidityRef = prev_valid and not na(prev_timestamp) and not na(prev_PDH) and not na(prev_PDL)
    if not hasLiquidityRef
        liquidityRefAt := na
        pdhTaken := false
        pdlTaken := false
        pdhSweepAt := na
        pdlSweepAt := na
        pdhSweepStage := 0
        pdlSweepStage := 0
        pdhSweepC1 := na
        pdlSweepC1 := na
    else
        if na(liquidityRefAt) or liquidityRefAt != prev_timestamp
            liquidityRefAt := prev_timestamp
            pdhTaken := false
            pdlTaken := false
            pdhSweepAt := na
            pdlSweepAt := na
            pdhSweepStage := 0
            pdlSweepStage := 0
            pdhSweepC1 := na
            pdlSweepC1 := na
        // No comparar velas del dia anterior con el nuevo PDH/PDL,
        // ni after-hours con niveles RTH. Solo M5 cerrada elegible.
        bool eligibleLiquidityBar = pdhInSession and not na(barT) and not na(barEnd) and barT >= liquidityRefAt
        if eligibleLiquidityBar
            int eventStage = c3Active and bucketH4 == c3Start ? 3 : c2Active and bucketH4 == c2Start ? 2 : 0
            int eventC1 = eventStage == 3 ? c1Start3 : eventStage == 2 ? c1Start2 : na
            if not pdhTaken and bH > prev_PDH
                pdhTaken := true
                pdhSweepAt := barEnd
                pdhSweepStage := eventStage
                pdhSweepC1 := eventC1
            if not pdlTaken and bL < prev_PDL
                pdlTaken := true
                pdlSweepAt := barEnd
                pdlSweepStage := eventStage
                pdlSweepC1 := eventC1

    // ---- M3B.2 INTEGRADO: PRIMERA BARRIDA M5 DE INSIDE BAR ----
    // M5 cerrada [1]; solo despues del cierre H4 de C2.
    // Se rastrean ambos extremos durante una MISMA H4 de manipulacion.
    if not na(barT) and not na(bucketH4) and array.size(insideCandidates) > 0
        for ic = 0 to array.size(insideCandidates) - 1
            InsideCandidate cand = array.get(insideCandidates, ic)
            if cand.state == "SEEKING" and barT >= cand.c2End
                bool brokeLow = bL < cand.c2Low
                bool brokeHigh = bH > cand.c2High
                if brokeLow or brokeHigh
                    cand.sweepAt := barEnd
                    cand.sweepH4Start := bucketH4
                    cand.sweptLow := brokeLow
                    cand.sweptHigh := brokeHigh
                    cand.sweepDir := brokeLow and not brokeHigh ? 1 : brokeHigh and not brokeLow ? -1 : 0
                    cand.state := brokeLow and brokeHigh ? "AMBIGUOUS" : "SWEPT"
                    // C1 y pivote son DEL Inside. Solo la primera M5
                    // barrida congela un pivote M15 anterior y confirmado.
                    bool freshInside15 = cand.sweepDir == 1 ? not na(lastHigh15At) and lastHigh15At < barT and barT - lastHigh15At <= maxPivotAge * 60000 : cand.sweepDir == -1 ? not na(lastLow15At) and lastLow15At < barT and barT - lastLow15At <= maxPivotAge * 60000 : false
                    cand.pivot15 := freshInside15 ? (cand.sweepDir == 1 ? lastHigh15 : lastLow15) : na
                    cand.pivot15At := freshInside15 ? (cand.sweepDir == 1 ? lastHigh15At : lastLow15At) : na
                    cand.mss15At := na
                    bool freshInside5 = cand.sweepDir == 1 ? not na(lastHigh5At) and not na(qaExtraM5SegmentStart) and lastHigh5At >= qaExtraM5SegmentStart and lastHigh5At < barT and barT - lastHigh5At <= maxPivotAge * 60000 : cand.sweepDir == -1 ? not na(lastLow5At) and not na(qaExtraM5SegmentStart) and lastLow5At >= qaExtraM5SegmentStart and lastLow5At < barT and barT - lastLow5At <= maxPivotAge * 60000 : false
                    cand.pivot5 := freshInside5 ? (cand.sweepDir == 1 ? lastHigh5 : lastLow5) : na
                    cand.pivot5At := freshInside5 ? (cand.sweepDir == 1 ? lastHigh5At : lastLow5At) : na
                    cand.mss5At := na
            else if cand.state == "SWEPT" and bucketH4 == cand.sweepH4Start
                cand.sweptLow := cand.sweptLow or bL < cand.c2Low
                cand.sweptHigh := cand.sweptHigh or bH > cand.c2High
